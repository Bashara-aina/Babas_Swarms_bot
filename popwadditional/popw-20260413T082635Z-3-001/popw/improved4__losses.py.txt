"""
Multi-Task Loss Functions
=========================
Detection : Focal Loss (14.7:1 imbalance)
Pose      : Wing Loss + confidence weighting (pseudo-label noise)
            Keypoints normalized to [0,1] for scale-invariant loss.
Activity  : Class-Balanced Focal Loss (2545:1 imbalance)
Multi-task: Kendall homoscedastic uncertainty weighting

Ablation support (Phase 2):
  - TRAIN_DET / TRAIN_POSE / TRAIN_ACT  : disable individual task losses
  - USE_KENDALL : toggle between learned uncertainty and equal weights

Author: Bashara
Date: February 2026
"""

import math
from typing import Dict, List, Tuple

import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from torchvision.ops import box_iou

import config as C


class FocalLoss(nn.Module):
    """Focal Loss for detection (Lin et al., 2017)."""

    def __init__(self, alpha=C.FOCAL_ALPHA, gamma=C.FOCAL_GAMMA,
                 pos_iou_thresh=0.5, neg_iou_thresh=0.4):
        super().__init__()
        self.alpha = alpha
        self.gamma = gamma
        self.pos_iou_thresh = pos_iou_thresh
        self.neg_iou_thresh = neg_iou_thresh

    def _match_anchors(self, anchors, gt_boxes, gt_labels):
        """Assign anchors to GTs. Returns matched labels (>=0 pos, -1 ignore, -2 bg) and boxes."""
        N = anchors.shape[0]
        device = anchors.device

        if gt_boxes.shape[0] == 0:
            return (torch.full((N,), -2, dtype=torch.long, device=device),
                    torch.zeros((N, 4), device=device))

        ious = box_iou(anchors, gt_boxes)
        max_iou, max_idx = ious.max(dim=1)

        labels = torch.full((N,), -1, dtype=torch.long, device=device)
        matched_boxes = gt_boxes[max_idx]
        labels[max_iou < self.neg_iou_thresh] = -2
        pos_mask = max_iou >= self.pos_iou_thresh
        labels[pos_mask] = gt_labels[max_idx[pos_mask]]

        for gi in range(gt_boxes.shape[0]):
            labels[ious[:, gi].argmax()] = gt_labels[gi]

        return labels, matched_boxes

    def _encode_boxes(self, anchors, gt_boxes):
        a_cx = (anchors[:, 0] + anchors[:, 2]) / 2
        a_cy = (anchors[:, 1] + anchors[:, 3]) / 2
        a_w = anchors[:, 2] - anchors[:, 0]
        a_h = anchors[:, 3] - anchors[:, 1]
        g_cx = (gt_boxes[:, 0] + gt_boxes[:, 2]) / 2
        g_cy = (gt_boxes[:, 1] + gt_boxes[:, 3]) / 2
        g_w = gt_boxes[:, 2] - gt_boxes[:, 0]
        g_h = gt_boxes[:, 3] - gt_boxes[:, 1]
        return torch.stack([
            (g_cx - a_cx) / (a_w + 1e-6),
            (g_cy - a_cy) / (a_h + 1e-6),
            torch.log((g_w + 1e-6) / (a_w + 1e-6)),
            torch.log((g_h + 1e-6) / (a_h + 1e-6)),
        ], dim=1)

    def forward(self, cls_preds, reg_preds, anchors, targets):
        """Returns cls_loss, reg_loss (both scalars)."""
        B = cls_preds.shape[0]
        device = cls_preds.device
        total_cls = torch.tensor(0.0, device=device)
        total_reg = torch.tensor(0.0, device=device)

        for i in range(B):
            gt_boxes = targets[i]['boxes'].to(device)
            gt_labels = targets[i]['labels'].to(device)
            matched_labels, matched_boxes = self._match_anchors(anchors, gt_boxes, gt_labels)

            pos_mask = matched_labels >= 0
            neg_mask = matched_labels == -2
            valid_mask = pos_mask | neg_mask
            num_pos = max(pos_mask.sum().item(), 1)

            cls_pred = cls_preds[i][valid_mask]
            cls_target = torch.zeros_like(cls_pred)
            if pos_mask.sum() > 0:
                pos_in_valid = pos_mask[valid_mask]
                cls_target[pos_in_valid, matched_labels[valid_mask][pos_in_valid]] = 1.0

            p = torch.sigmoid(cls_pred)
            ce = F.binary_cross_entropy_with_logits(cls_pred, cls_target, reduction='none')
            p_t = p * cls_target + (1 - p) * (1 - cls_target)
            alpha_t = self.alpha * cls_target + (1 - self.alpha) * (1 - cls_target)
            total_cls = total_cls + (alpha_t * (1 - p_t) ** self.gamma * ce).sum() / num_pos

            if pos_mask.sum() > 0:
                deltas = self._encode_boxes(anchors[pos_mask], matched_boxes[pos_mask])
                total_reg = total_reg + F.smooth_l1_loss(
                    reg_preds[i][pos_mask], deltas, beta=0.11, reduction='sum'
                ) / num_pos

        return total_cls / B, total_reg / B


class WingLoss(nn.Module):
    """Wing Loss for keypoint regression (Feng et al., 2018).

    Coordinates are normalized to [0, 1] before loss computation so that
    the loss magnitude is independent of image resolution. WING_OMEGA and
    WING_EPSILON in config.py are tuned for this normalized range.
    """

    def __init__(self, omega=C.WING_OMEGA, epsilon=C.WING_EPSILON):
        super().__init__()
        self.omega = omega
        self.epsilon = epsilon
        self.C = omega - omega * math.log(1 + omega / epsilon)

    def forward(self, pred, gt, visibility, confidence):
        """pred/gt: [B,17,2], visibility/confidence: [B,17]."""
        # Normalize pixel coordinates to [0, 1] for scale-invariant loss
        scale = torch.tensor(
            [C.IMG_WIDTH, C.IMG_HEIGHT],
            device=pred.device, dtype=pred.dtype,
        )
        pred_norm = pred / scale
        gt_norm = gt / scale

        abs_diff = torch.abs(pred_norm - gt_norm)
        loss = torch.where(
            abs_diff < self.omega,
            self.omega * torch.log(1 + abs_diff / self.epsilon),
            abs_diff - self.C,
        )
        # Visibility (binary gate: conf > threshold) and confidence (soft quality
        # weight within the visible set) are both applied intentionally.
        # Down-weights uncertain pseudo-labels without excluding them: conf=0.15
        # gets 15% of the gradient of conf=1.0. Not double-weighting — gate then scale.
        loss_per_kpt = loss.sum(dim=-1) * confidence * visibility
        return loss_per_kpt.sum() / visibility.sum().clamp(min=1.0)


class ClassBalancedFocalLoss(nn.Module):
    """CB Focal Loss (Cui et al., 2019) for 2545:1 activity imbalance."""

    def __init__(self, num_classes=C.NUM_ACT_CLASSES, beta=C.CB_BETA, gamma=C.CB_GAMMA):
        super().__init__()
        self.num_classes = num_classes
        self.beta = beta
        self.gamma = gamma
        self.register_buffer('class_weights', torch.ones(num_classes))

    def set_class_counts(self, counts):
        counts = np.array(counts, dtype=np.float64)
        effective = np.where(
            counts > 0,
            (1.0 - np.power(self.beta, counts)) / (1.0 - self.beta),
            1.0,
        )
        weights = 1.0 / np.maximum(effective, 1e-8)
        weights = weights / weights.sum() * self.num_classes
        # Use .data.copy_() to update the registered buffer in-place.
        # Plain assignment (self.class_weights = ...) replaces the buffer
        # with a plain tensor that won't be moved by .to(device).
        self.class_weights.data.copy_(
            torch.tensor(weights, dtype=torch.float32)
        )

    def forward(self, logits, targets):
        device = logits.device
        w = self.class_weights.to(device)[targets]
        ce = F.cross_entropy(logits, targets, reduction='none')
        p_t = torch.exp(-ce)
        return (w * (1 - p_t) ** self.gamma * ce).mean()


class MultiTaskLoss(nn.Module):
    """
    Kendall homoscedastic uncertainty weighting (Kendall et al., 2018).
    L = sum_t [ exp(-s_t) * L_t + s_t ] where s_t = log(sigma_t^2)

    Ablation support:
      - train_det/train_pose/train_act : zero out disabled task losses
      - use_kendall : when False, use equal weights (1.0) instead of learned

    log_var initialization notes:
      - log_var_det  = 0.0  (precision=1.0, neutral start)
      - log_var_pose = -1.0 (precision~2.7x higher at init; compensates for
                             Wing loss on [0,1] coords being 10-50x smaller
                             than focal det/act losses at initialization)
      - log_var_act  = 0.0  (precision=1.0, neutral start)

    Kendall clamp range [-4, 2]:
      - exp(-(-4)) = exp(4) ~ 55   (max precision; safe in FP16 max 65504)
      - exp(-(2))  = exp(-2) ~ 0.14 (min precision; task can be down-weighted)
      - Widened from [-3, 1] after initial convergence confirmed stable.
    """

    def __init__(self, num_classes=C.NUM_ACT_CLASSES,
                 train_det=C.TRAIN_DET, train_pose=C.TRAIN_POSE,
                 train_act=C.TRAIN_ACT, use_kendall=C.USE_KENDALL):
        super().__init__()
        self.train_det = train_det
        self.train_pose = train_pose
        self.train_act = train_act
        self.use_kendall = use_kendall

        # Detection and activity start neutral (precision=1.0)
        self.log_var_det = nn.Parameter(torch.zeros(1))
        # Pose gets higher initial precision to compensate for smaller loss scale.
        # Wing loss on normalized [0,1] coords is ~10-50x smaller than det/act
        # losses at initialization. Without this, Kendall takes 10-20 epochs
        # to compensate, causing the pose head to lag.
        self.log_var_pose = nn.Parameter(torch.tensor([-1.0]))
        self.log_var_act = nn.Parameter(torch.zeros(1))

        self.det_loss = FocalLoss()
        self.pose_loss = WingLoss()
        self.act_loss = ClassBalancedFocalLoss(num_classes=num_classes)

        # Activity warmup: ramp activity loss weight from 0→1 over first
        # ACT_WARMUP_EPOCHS epochs. Gives the pose head time to produce
        # meaningful keypoints before FiLM conditioning kicks in.
        self._act_warmup_epochs = int(getattr(C, 'ACT_WARMUP_EPOCHS', 5))
        self._current_epoch = 0  # set externally via set_epoch()

    def set_class_counts(self, counts):
        self.act_loss.set_class_counts(counts)

    def set_epoch(self, epoch: int):
        """Call at the start of each epoch to update activity warmup ramp."""
        self._current_epoch = epoch

    def forward(self, outputs, targets):
        device = outputs['cls_preds'].device
        zero = torch.tensor(0.0, device=device)

        # Compute raw task losses (or zero if disabled)
        if self.train_det:
            cls_loss, reg_loss = self.det_loss(
                outputs['cls_preds'], outputs['reg_preds'],
                outputs['anchors'], targets['detection'],
            )
            loss_det = cls_loss + reg_loss
        else:
            cls_loss, reg_loss, loss_det = zero, zero, zero

        if self.train_pose:
            loss_pose = self.pose_loss(
                outputs['keypoints'], targets['keypoints'],
                targets['visibility'], targets['kpt_confidence'],
            )
        else:
            loss_pose = zero

        if self.train_act:
            loss_act = self.act_loss(outputs['act_logits'], targets['activity'])
        else:
            loss_act = zero

        # Activity warmup ramp: 0→1 over first ACT_WARMUP_EPOCHS epochs.
        # Gives pose head time to produce meaningful keypoints before
        # activity gradients (and FiLM conditioning) kick in.
        act_ramp = min(1.0, self._current_epoch / max(self._act_warmup_epochs, 1))
        loss_act = loss_act * act_ramp

        # Combine with Kendall weighting or equal weights
        if self.use_kendall:
            # Clamp log_var to [-4, 2]:
            #   exp(4)  ~ 55   max precision (safe in FP16, well below 65504)
            #   exp(-2) ~ 0.14 min precision (task can be substantially down-weighted)
            # Widened from [-3, 1] to give the model more room to balance tasks
            # after initial convergence is confirmed stable.
            lv_d = self.log_var_det.clamp(-4.0, 2.0)
            lv_p = self.log_var_pose.clamp(-4.0, 2.0)
            lv_a = self.log_var_act.clamp(-4.0, 2.0)

            prec_d = torch.exp(-lv_d)
            prec_p = torch.exp(-lv_p)
            prec_a = torch.exp(-lv_a)

            total = torch.tensor(0.0, device=device)
            if self.train_det:
                total = total + prec_d * loss_det + lv_d
            if self.train_pose:
                total = total + prec_p * loss_pose + lv_p
            if self.train_act:
                total = total + prec_a * loss_act + lv_a
            total = total.squeeze()
        else:
            # Equal weighting -- log_var params are not used
            prec_d = prec_p = prec_a = torch.tensor(1.0, device=device)
            total = loss_det + loss_pose + loss_act

        with torch.no_grad():
            wd, wp, wa = prec_d.item(), prec_p.item(), prec_a.item()
            ws = wd + wp + wa + 1e-8

        # Report both raw and clamped log_var for diagnostics
        loss_dict = {
            'total': total.item(), 'det_cls': cls_loss.item(),
            'det_reg': reg_loss.item(), 'det': loss_det.item(),
            'pose': loss_pose.item(), 'activity': loss_act.item(),
            'w_det': wd / ws, 'w_pose': wp / ws, 'w_act': wa / ws,
            'log_var_det': self.log_var_det.item(),
            'log_var_pose': self.log_var_pose.item(),
            'log_var_act': self.log_var_act.item(),
        }
        return total, loss_dict
