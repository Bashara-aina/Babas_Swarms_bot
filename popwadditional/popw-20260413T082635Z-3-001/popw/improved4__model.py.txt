"""
Multi-Task IKEA Assembly Model
===============================
ResNet-50-FPN backbone with three task-specific heads:
  1. Detection  : RetinaNet-style anchor-based (7 furniture part classes)
  2. Pose       : Heatmap-based with soft-argmax (17 COCO keypoints)
  3. Activity   : GAP + FC classifier (33 assembly actions)

Optional cross-task modules (Phase 2, gated by config flags):
  - PoseFiLMModule : Pose-conditioned FiLM modulation on C5 before activity head

Fixes (2026-03-10):
  - ActivityHead: removed broken p4=None branch that produced dim mismatch
    (Linear expected 2048+256=2304, got 2048). The head always fuses C5+P4.
    A dedicated C5-only constructor arg is provided for explicit ablation use.
  - PoseHead.soft_argmax: fixed edge-pixel bias. Scaling now uses
    (IMG_W-1)/(hm_w-1) instead of IMG_W/hm_w, eliminating the ~8px
    systematic underestimation at image boundaries.
  - AnchorGenerator: fixed ratio/scale loop order to match standard RetinaNet
    (ratios outer, scales inner). Ensures anchor ordering is compatible with
    pretrained RetinaNet checkpoints and published implementations.

Author: Bashara
Date: February 2026 | Audited: March 2026
"""

import math
from typing import Dict, List, Optional, Tuple

import torch
import torch.nn as nn
import torch.nn.functional as F
from torchvision.models import resnet50, ResNet50_Weights

import config as C

# Backbone / Neck

class FPN(nn.Module):
    """Standard FPN. Takes C3, C4, C5 -> P3, P4, P5, P6, P7. All 256 channels."""

    def __init__(self, in_channels_list: List[int], out_channels: int = 256):
        super().__init__()
        self.lateral_c3 = nn.Conv2d(in_channels_list[0], out_channels, 1)
        self.lateral_c4 = nn.Conv2d(in_channels_list[1], out_channels, 1)
        self.lateral_c5 = nn.Conv2d(in_channels_list[2], out_channels, 1)
        self.smooth_p3  = nn.Conv2d(out_channels, out_channels, 3, padding=1)
        self.smooth_p4  = nn.Conv2d(out_channels, out_channels, 3, padding=1)
        self.smooth_p5  = nn.Conv2d(out_channels, out_channels, 3, padding=1)
        self.p6_conv    = nn.Conv2d(in_channels_list[2], out_channels, 3, stride=2, padding=1)
        self.p7_conv    = nn.Conv2d(out_channels, out_channels, 3, stride=2, padding=1)
        self._init_weights()

    def _init_weights(self):
        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_uniform_(m.weight, a=1)
                if m.bias is not None:
                    nn.init.zeros_(m.bias)

    def forward(self, c3, c4, c5):
        p5 = self.lateral_c5(c5)
        p4 = self.lateral_c4(c4) + F.interpolate(p5, size=c4.shape[2:], mode='nearest')
        p3 = self.lateral_c3(c3) + F.interpolate(p4, size=c3.shape[2:], mode='nearest')
        p3 = self.smooth_p3(p3)
        p4 = self.smooth_p4(p4)
        p5 = self.smooth_p5(p5)
        p6 = self.p6_conv(c5)
        p7 = self.p7_conv(F.relu(p6))
        return {'p3': p3, 'p4': p4, 'p5': p5, 'p6': p6, 'p7': p7}


class AnchorGenerator(nn.Module):
    """
    Generate anchors for RetinaNet. 3 ratios x 3 scales = 9 per location.

    FIXED (2026-03-10): Loop order corrected to ratios-outer / scales-inner,
    matching the standard RetinaNet paper and torchvision implementation.
    Previous code had scales-outer / ratios-inner which produced differently
    ordered anchor tensors -- incompatible with pretrained RetinaNet weights.
    """

    def __init__(
        self,
        sizes:   Tuple[int, ...] = (32, 64, 128, 256, 512),
        ratios:  Tuple[float, ...] = (0.5, 1.0, 2.0),
        scales:  Tuple[float, ...] = (1.0, 2 ** (1 / 3), 2 ** (2 / 3)),
    ):
        super().__init__()
        self.sizes   = sizes
        self.ratios  = ratios
        self.scales  = scales
        self.num_anchors = len(ratios) * len(scales)  # 9

    def forward(self, feature_maps: Dict[str, torch.Tensor]) -> torch.Tensor:
        device = next(iter(feature_maps.values())).device
        all_anchors = []
        fpn_keys = ['p3', 'p4', 'p5', 'p6', 'p7']
        strides   = [8,   16,   32,   64,   128]

        for level_idx, (key, stride) in enumerate(zip(fpn_keys, strides)):
            feat = feature_maps[key]
            h, w = feat.shape[2:]
            base_size = self.sizes[level_idx]

            # FIXED: ratios outer, scales inner -- matches RetinaNet paper
            cell_anchors = []
            for ratio in self.ratios:
                for scale in self.scales:
                    s  = base_size * scale
                    aw = s * math.sqrt(ratio)
                    ah = s / math.sqrt(ratio)
                    cell_anchors.append([-aw / 2, -ah / 2, aw / 2, ah / 2])
            cell_anchors = torch.tensor(
                cell_anchors, device=device, dtype=torch.float32
            )  # [9, 4]

            shifts_x = (torch.arange(w, device=device) + 0.5) * stride
            shifts_y = (torch.arange(h, device=device) + 0.5) * stride
            shift_y, shift_x = torch.meshgrid(shifts_y, shifts_x, indexing='ij')
            shifts = torch.stack([
                shift_x.flatten(), shift_y.flatten(),
                shift_x.flatten(), shift_y.flatten(),
            ], dim=1)  # [H*W, 4]

            anchors = shifts.unsqueeze(1) + cell_anchors.unsqueeze(0)  # [H*W, 9, 4]
            all_anchors.append(anchors.reshape(-1, 4))

        return torch.cat(all_anchors, dim=0)

class DetectionHead(nn.Module):
    """RetinaNet classification + regression subnets, shared across FPN levels."""

    def __init__(self, in_channels: int = 256,
                 num_classes: int = C.NUM_DET_CLASSES,
                 num_anchors: int = 9):
        super().__init__()
        self.num_classes = num_classes
        self.num_anchors = num_anchors

        cls_layers = []
        for _ in range(4):
            cls_layers.extend([
                nn.Conv2d(in_channels, in_channels, 3, padding=1),
                nn.ReLU(True),
            ])
        self.cls_subnet = nn.Sequential(*cls_layers)
        self.cls_score  = nn.Conv2d(in_channels, num_anchors * num_classes, 3, padding=1)

        reg_layers = []
        for _ in range(4):
            reg_layers.extend([
                nn.Conv2d(in_channels, in_channels, 3, padding=1),
                nn.ReLU(True),
            ])
        self.reg_subnet = nn.Sequential(*reg_layers)
        self.reg_pred   = nn.Conv2d(in_channels, num_anchors * 4, 3, padding=1)

        self._init_weights()

    def _init_weights(self):
        for modules in [self.cls_subnet, self.reg_subnet]:
            for m in modules.modules():
                if isinstance(m, nn.Conv2d):
                    nn.init.normal_(m.weight, std=0.01)
                    nn.init.zeros_(m.bias)
        pi = 0.01
        nn.init.normal_(self.cls_score.weight, std=0.01)
        nn.init.constant_(self.cls_score.bias, -math.log((1 - pi) / pi))
        nn.init.normal_(self.reg_pred.weight, std=0.01)
        nn.init.zeros_(self.reg_pred.bias)

    def forward(self, features: Dict[str, torch.Tensor]):
        cls_all, reg_all = [], []
        for key in ('p3', 'p4', 'p5', 'p6', 'p7'):
            feat = features[key]
            B, _, H, W = feat.shape
            cls_out = self.cls_score(self.cls_subnet(feat))
            cls_out = cls_out.permute(0, 2, 3, 1).reshape(
                B, H * W * self.num_anchors, self.num_classes
            )
            cls_all.append(cls_out)
            reg_out = self.reg_pred(self.reg_subnet(feat))
            reg_out = reg_out.permute(0, 2, 3, 1).reshape(
                B, H * W * self.num_anchors, 4
            )
            reg_all.append(reg_out)
        return torch.cat(cls_all, dim=1), torch.cat(reg_all, dim=1)


class PoseHead(nn.Module):
    """
    Heatmap-based 2D pose head. Deconv upsampling + soft-argmax.

    FIXED (2026-03-10): soft_argmax coordinate scaling now uses
        (IMG_W - 1) / (hm_w - 1)  and  (IMG_H - 1) / (hm_h - 1)
    instead of IMG_W / hm_w (IMG_H / hm_h). The old formulation mapped
    heatmap index 0..hm_w-1 onto pixel range 0..IMG_W*(hm_w-1)/hm_w,
    which systematically underestimated edge keypoints by ~8 pixels at
    640px width. The corrected formulation maps 0 -> 0 and hm_w-1 -> IMG_W-1
    exactly, eliminating the boundary bias.
    """

    def __init__(self, in_channels: int = 256,
                 num_keypoints: int = C.NUM_KEYPOINTS,
                 heatmap_size:  int = 64):
        super().__init__()
        self.num_keypoints = num_keypoints
        self.heatmap_size  = heatmap_size

        # Single ConvTranspose2d stride=2 (P3 60x80 -> 120x160) then interpolate
        # to heatmap_size.  GroupNorm replaces BatchNorm2d: no running statistics
        # that accumulate NaN, and batch-size-independent (batch=6 too small for BN).
        self.deconv = nn.Sequential(
            nn.ConvTranspose2d(in_channels, 256, 4, stride=2, padding=1),
            nn.GroupNorm(32, 256),
            nn.ReLU(True),
        )
        self.heatmap_conv = nn.Conv2d(256, num_keypoints, 1)
        self._init_weights()

    def _init_weights(self):
        for m in self.deconv.modules():
            if isinstance(m, nn.ConvTranspose2d):
                nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='relu')
                if m.bias is not None:
                    nn.init.zeros_(m.bias)
            elif isinstance(m, nn.GroupNorm):
                nn.init.ones_(m.weight)
                nn.init.zeros_(m.bias)
        nn.init.normal_(self.heatmap_conv.weight, std=0.001)
        nn.init.zeros_(self.heatmap_conv.bias)

    def soft_argmax(self, heatmaps: torch.Tensor) -> torch.Tensor:
        """
        Convert heatmaps [B, K, H, W] to heatmap-space coordinates [B, K, 2].

        FIXED: uses (W-1) and (H-1) as the coordinate range so that the
        maximum heatmap index maps exactly to the last pixel position.
        The old IMG_W/hm_w scaling created a systematic ~8px boundary bias.
        """
        B, K, H, W = heatmaps.shape
        hm_flat = heatmaps.reshape(B, K, -1)
        hm_soft = F.softmax(hm_flat, dim=-1).reshape(B, K, H, W)
        device  = heatmaps.device

        # Coordinate grids in heatmap pixel space: 0 .. W-1, 0 .. H-1
        x_coords = torch.arange(W, device=device, dtype=torch.float32)  # [W]
        y_coords = torch.arange(H, device=device, dtype=torch.float32)  # [H]

        # sum(dim=2) collapses H, leaving [B, K, W] to weight against x_coords
        exp_x = (hm_soft.sum(dim=2) * x_coords).sum(dim=-1)  # [B, K]
        # sum(dim=3) collapses W, leaving [B, K, H] to weight against y_coords
        exp_y = (hm_soft.sum(dim=3) * y_coords).sum(dim=-1)  # [B, K]

        coords = torch.stack([exp_x, exp_y], dim=-1)           # [B, K, 2]
        return coords.nan_to_num(0.0)  # guard degenerate heatmaps

    def forward(self, p3: torch.Tensor):
        x    = self.deconv(p3)
        hm_h = self.heatmap_size
        hm_w = int(self.heatmap_size * C.IMG_WIDTH / C.IMG_HEIGHT)  # = 85
        x    = F.interpolate(x, size=(hm_h, hm_w), mode='bilinear', align_corners=False)
        heatmaps = self.heatmap_conv(x)   # [B, K, hm_h, hm_w]

        coords_hm = self.soft_argmax(heatmaps)  # [B, K, 2]  in heatmap space

        # FIXED: scale from heatmap coords to image pixel coords using
        # (IMG_DIM - 1) / (hm_DIM - 1) so endpoints map exactly.
        # Old: coords * IMG_W / hm_w  -> max = (hm_w-1) * IMG_W/hm_w < IMG_W-1
        # New: coords * (IMG_W-1) / (hm_w-1) -> max = IMG_W-1  (exact)
        scale_x = (C.IMG_WIDTH  - 1) / max(hm_w - 1, 1)
        scale_y = (C.IMG_HEIGHT - 1) / max(hm_h - 1, 1)

        coords = coords_hm.clone()
        coords[:, :, 0] = coords_hm[:, :, 0] * scale_x
        coords[:, :, 1] = coords_hm[:, :, 1] * scale_y

        return heatmaps, coords


class ActivityHead(nn.Module):
    """
    GAP on C5 + P4 features -> FC classifier.

    Fusing P4 (stride=16, 40x30) with C5 (stride=32, 20x15) adds hand/tool
    spatial detail critical for 33-class assembly action discrimination.

    FIXED (2026-03-10): Removed the broken `p4=None` fallback branch.
    The old code constructed the Linear layer with in_features=2048+256=2304
    but the p4=None path only concatenated C5 (2048 dims), causing a
    guaranteed RuntimeError on the Linear forward pass.

    The head now always requires p4.  For ablation experiments that need
    C5-only mode, instantiate with `fuse_p4=False` which sets in_features=2048.
    """

    def __init__(
        self,
        c5_channels:  int = 2048,
        p4_channels:  int = 256,
        hidden_dim:   int = 512,
        num_classes:  int = C.NUM_ACT_CLASSES,
        dropout:      float = 0.3,
        fuse_p4:      bool = True,        # set False for C5-only ablation
        use_residual: bool = True,
    ):
        super().__init__()
        self.fuse_p4 = fuse_p4
        self.use_residual = use_residual
        in_features = (c5_channels + p4_channels) if fuse_p4 else c5_channels

        self.pool_c5 = nn.AdaptiveAvgPool2d(1)
        self.pool_p4 = nn.AdaptiveAvgPool2d(1) if fuse_p4 else None
        if self.use_residual:
            self.fc1 = nn.Sequential(
                nn.Linear(in_features, hidden_dim),
                nn.BatchNorm1d(hidden_dim),
                nn.ReLU(True),
                nn.Dropout(dropout),
            )
            self.bottleneck = nn.Sequential(
                nn.Linear(hidden_dim, hidden_dim // 2),
                nn.BatchNorm1d(hidden_dim // 2),
                nn.ReLU(True),
                nn.Dropout(dropout),
            )
            self.fc2 = nn.Sequential(
                nn.Linear(hidden_dim // 2, hidden_dim),
                nn.BatchNorm1d(hidden_dim),
                nn.ReLU(True),
                nn.Dropout(dropout),
            )
            self.residual_projection = nn.Linear(in_features, hidden_dim)
            self.classifier = nn.Linear(hidden_dim, num_classes)
        else:
            self.fc = nn.Sequential(
                nn.Linear(in_features, hidden_dim),
                nn.ReLU(True),
                nn.Dropout(dropout),
                nn.Linear(hidden_dim, num_classes),
            )

    def forward(self, c5: torch.Tensor,
                p4: Optional[torch.Tensor] = None) -> torch.Tensor:
        """
        Args:
            c5 : [B, 2048, H, W]  -- required
            p4 : [B, 256,  H, W]  -- required when fuse_p4=True (default)

        Raises:
            ValueError: if fuse_p4=True but p4 is None (prevents silent
                        dimension mismatch inside the Linear layer).
        """
        if self.fuse_p4:
            if p4 is None:
                raise ValueError(
                    'ActivityHead requires p4 when fuse_p4=True. '
                    'Pass pyramid["p4"] or construct with fuse_p4=False.'
                )
            x = torch.cat([self.pool_c5(c5), self.pool_p4(p4)], dim=1)  # [B, 2304, 1, 1]
        else:
            x = self.pool_c5(c5)  # [B, 2048, 1, 1]

        x = x.flatten(1)
        if self.use_residual:
            residual = self.residual_projection(x)
            x = self.fc1(x)
            x = self.bottleneck(x)
            x = self.fc2(x)
            x = x + residual
            return self.classifier(x)
        return self.fc(x)

class PoseFiLMModule(nn.Module):
    """
    Pose-conditioned FiLM modulation (Perez et al., 2018).

    Modulates C5 feature maps using pose keypoint coordinates and
    per-keypoint confidence scores before the activity head.

    Input:
        features   : [B, 2048, H, W]  -- C5 feature map
        keypoints  : [B, 17, 2]       -- predicted (x, y) pixel coords
        confidence : [B, 17]           -- per-keypoint confidence

    Output:
        modulated  : [B, 2048, H, W]  -- gamma * features + beta
    """

    def __init__(self, pose_dim: int = C.NUM_KEYPOINTS * 3,
                 feat_channels: int = 2048):
        super().__init__()
        self.gamma_net = nn.Sequential(
            nn.Linear(pose_dim, 512), nn.ReLU(True),
            nn.Linear(512, feat_channels),
        )
        self.beta_net = nn.Sequential(
            nn.Linear(pose_dim, 512), nn.ReLU(True),
            nn.Linear(512, feat_channels),
        )
        self._init_weights()

    def _init_weights(self):
        # Initialize gamma ~ 1, beta ~ 0 so module starts as identity.
        for net in [self.gamma_net, self.beta_net]:
            for m in net.modules():
                if isinstance(m, nn.Linear):
                    nn.init.normal_(m.weight, std=0.01)
                    nn.init.zeros_(m.bias)
        nn.init.ones_(self.gamma_net[-1].bias)  # bias final gamma toward 1

    def forward(self, features: torch.Tensor,
                keypoints: torch.Tensor,
                confidence: torch.Tensor) -> torch.Tensor:
        scale = torch.tensor(
            [C.IMG_WIDTH, C.IMG_HEIGHT],
            device=keypoints.device, dtype=keypoints.dtype,
        )
        kpts_norm = keypoints / scale                          # [B, 17, 2] in [0,1]
        pose_flat = torch.cat(
            [kpts_norm.flatten(1), confidence], dim=1
        )                                                      # [B, 51]

        # Bound gamma via 1+tanh to keep it in (0, 2), centered at identity (1.0).
        # Prevents gamma from exploding/collapsing on early training or outlier pose.
        # Consistent with AdaLN-Zero initialization strategy and FiLM literature.
        # Beta remains unbounded (shift is safe, scale is dangerous).
        gamma_raw = self.gamma_net(pose_flat)
        gamma = (1.0 + torch.tanh(gamma_raw)).unsqueeze(-1).unsqueeze(-1)  # [B, 2048, 1, 1]
        beta  = self.beta_net(pose_flat).unsqueeze(-1).unsqueeze(-1)       # [B, 2048, 1, 1]

        return gamma * features + beta


class PoseCrossAttentionModule(nn.Module):
    """Pose/object cross-attention modulation on C5 features."""

    def __init__(
        self,
        feat_channels: int = 2048,
        embed_dim: int = 256,
        num_heads: int = 8,
    ):
        super().__init__()
        self.query_proj = nn.Linear(feat_channels, embed_dim)
        self.pose_proj = nn.Linear(3, embed_dim)
        self.obj_proj = nn.Linear(4 + C.NUM_DET_CLASSES, embed_dim)

        self.attn = nn.MultiheadAttention(
            embed_dim=embed_dim,
            num_heads=num_heads,
            batch_first=True,
        )
        self.norm1 = nn.LayerNorm(embed_dim)
        self.ffn = nn.Sequential(
            nn.Linear(embed_dim, embed_dim * 4),
            nn.GELU(),
            nn.Linear(embed_dim * 4, embed_dim),
        )
        self.norm2 = nn.LayerNorm(embed_dim)
        self.out_proj = nn.Linear(embed_dim, feat_channels)

    def forward(
        self,
        features: torch.Tensor,
        keypoints: torch.Tensor,
        kpt_confidence: torch.Tensor,
        boxes: Optional[torch.Tensor] = None,
        box_logits: Optional[torch.Tensor] = None,
    ) -> torch.Tensor:
        bsz, channels, height, width = features.shape

        q = features.permute(0, 2, 3, 1).reshape(bsz, height * width, channels)
        q = self.query_proj(q)

        scale = torch.tensor(
            [C.IMG_WIDTH, C.IMG_HEIGHT],
            device=keypoints.device,
            dtype=keypoints.dtype,
        )
        kpts_norm = keypoints / scale
        pose_tokens = torch.cat([kpts_norm, kpt_confidence.unsqueeze(-1)], dim=-1)
        kv = [self.pose_proj(pose_tokens)]

        if boxes is not None and box_logits is not None and boxes.numel() > 0:
            boxes_norm = boxes.clone()
            boxes_norm[..., [0, 2]] /= float(C.IMG_WIDTH)
            boxes_norm[..., [1, 3]] /= float(C.IMG_HEIGHT)
            obj_tokens = torch.cat([boxes_norm, torch.sigmoid(box_logits)], dim=-1)
            kv.append(self.obj_proj(obj_tokens))

        kv_tokens = torch.cat(kv, dim=1)
        attn_out, _ = self.attn(q, kv_tokens, kv_tokens)
        x = self.norm1(q + attn_out)
        x = self.norm2(x + self.ffn(x))
        x = self.out_proj(x).reshape(bsz, height, width, channels).permute(0, 3, 1, 2)
        return x + features

class MultiTaskIKEA(nn.Module):
    """
    Unified multi-task model for IKEA assembly.
    Backbone: ResNet-50 (ImageNet pretrained, frozen BN)
    Neck    : FPN (P3-P7)
    Heads   : Detection (RetinaNet), Pose (Heatmap), Activity (FC)
    Optional: PoseFiLM modulation (enabled via use_film flag)
    """

    def __init__(self, pretrained: bool = True, use_film: bool = C.USE_FILM):
        super().__init__()

        backbone = resnet50(
            weights=ResNet50_Weights.IMAGENET1K_V2 if pretrained else None
        )

        self.layer0 = nn.Sequential(
            backbone.conv1, backbone.bn1, backbone.relu, backbone.maxpool
        )
        self.layer1 = backbone.layer1   # C2: 256ch,  stride 4
        self.layer2 = backbone.layer2   # C3: 512ch,  stride 8
        self.layer3 = backbone.layer3   # C4: 1024ch, stride 16
        self.layer4 = backbone.layer4   # C5: 2048ch, stride 32
        self._freeze_bn()

        self.fpn            = FPN([512, 1024, 2048], 256)
        self.detection_head = DetectionHead(256, C.NUM_DET_CLASSES)
        self.pose_head      = PoseHead(256, C.NUM_KEYPOINTS)
        # fuse_p4=True: always requires p4 -- eliminates the silent dim mismatch
        self.activity_head  = ActivityHead(2048, 256, 512, C.NUM_ACT_CLASSES,
                                           fuse_p4=True)
        self.anchor_gen     = AnchorGenerator()

        self.use_film = use_film
        if self.use_film:
            self.use_transformer = bool(getattr(C, 'USE_TRANSFORMER', False))
            if self.use_transformer:
                self.film = PoseCrossAttentionModule(feat_channels=2048)
            else:
                self.film = PoseFiLMModule(
                    pose_dim=C.NUM_KEYPOINTS * 3, feat_channels=2048
                )
        else:
            self.use_transformer = False

    @staticmethod
    def _decode_boxes(anchors: torch.Tensor, deltas: torch.Tensor) -> torch.Tensor:
        a_cx = (anchors[:, 0] + anchors[:, 2]) / 2
        a_cy = (anchors[:, 1] + anchors[:, 3]) / 2
        a_w = anchors[:, 2] - anchors[:, 0]
        a_h = anchors[:, 3] - anchors[:, 1]

        dx = deltas[:, 0]
        dy = deltas[:, 1]
        dw = deltas[:, 2].clamp(-4, 4)
        dh = deltas[:, 3].clamp(-4, 4)

        pred_w = torch.exp(dw) * a_w
        pred_h = torch.exp(dh) * a_h
        pred_cx = dx * a_w + a_cx
        pred_cy = dy * a_h + a_cy

        return torch.stack([
            pred_cx - pred_w / 2,
            pred_cy - pred_h / 2,
            pred_cx + pred_w / 2,
            pred_cy + pred_h / 2,
        ], dim=1)

    def _freeze_bn(self):
        for module in [
            self.layer0, self.layer1, self.layer2, self.layer3, self.layer4
        ]:
            for m in module.modules():
                if isinstance(m, (nn.BatchNorm2d, nn.SyncBatchNorm)):
                    m.eval()
                    for p in m.parameters():
                        p.requires_grad = False

    def train(self, mode: bool = True):
        super().train(mode)
        if mode:
            self._freeze_bn()
        return self

    def forward(self, images: torch.Tensor) -> Dict[str, torch.Tensor]:
        x  = self.layer0(images)
        c2 = self.layer1(x)
        c3 = self.layer2(c2)
        c4 = self.layer3(c3)
        c5 = self.layer4(c4)

        pyramid                  = self.fpn(c3, c4, c5)
        cls_preds, reg_preds     = self.detection_head(pyramid)
        anchors                  = self.anchor_gen(pyramid)
        heatmaps, keypoints      = self.pose_head(pyramid['p3'])

        # Activity head -- optionally modulated by pose via FiLM
        if self.use_film:
            with torch.no_grad():
                B, K, H, W = heatmaps.shape
                hm_flat    = heatmaps.reshape(B, K, -1)
                confidence = hm_flat.max(dim=-1).values.sigmoid().nan_to_num(0.5)

            if self.use_transformer:
                topk = min(20, cls_preds.shape[1])
                cls_conf = torch.sigmoid(cls_preds)
                anchor_conf = cls_conf.max(dim=-1).values
                top_idx = anchor_conf.topk(k=topk, dim=1).indices
                batch_idx = torch.arange(B, device=images.device).unsqueeze(1)

                top_deltas = reg_preds[batch_idx, top_idx]
                top_logits = cls_preds[batch_idx, top_idx]
                anchors_b = anchors.unsqueeze(0).expand(B, -1, 4)
                top_anchors = anchors_b[batch_idx, top_idx]

                top_boxes = self._decode_boxes(
                    top_anchors.reshape(-1, 4),
                    top_deltas.reshape(-1, 4),
                ).reshape(B, topk, 4)
                top_boxes[..., [0, 2]] = top_boxes[..., [0, 2]].clamp(0, C.IMG_WIDTH)
                top_boxes[..., [1, 3]] = top_boxes[..., [1, 3]].clamp(0, C.IMG_HEIGHT)

                c5_mod = self.film(
                    c5,
                    keypoints.nan_to_num(0.0),
                    confidence,
                    boxes=top_boxes,
                    box_logits=top_logits,
                )
            else:
                c5_mod = self.film(
                    c5, keypoints.nan_to_num(0.0), confidence
                )
            act_logits = self.activity_head(c5_mod, pyramid['p4'])
        else:
            # fuse_p4=True: always pass p4 (raises ValueError otherwise)
            act_logits = self.activity_head(c5, pyramid['p4'])

        return {
            'cls_preds':  cls_preds,
            'reg_preds':  reg_preds,
            'anchors':    anchors,
            'heatmaps':   heatmaps,
            'keypoints':  keypoints,
            'act_logits': act_logits,
        }


def count_parameters(model: MultiTaskIKEA) -> Dict[str, int]:
    components = {
        'backbone':  [model.layer0, model.layer1, model.layer2,
                      model.layer3, model.layer4],
        'fpn':       [model.fpn],
        'detection': [model.detection_head],
        'pose':      [model.pose_head],
        'activity':  [model.activity_head],
    }
    if hasattr(model, 'film'):
        components['film'] = [model.film]
    result = {}
    total  = 0
    for name, modules in components.items():
        count = sum(
            p.numel() for m in modules for p in m.parameters()
            if p.requires_grad
        )
        result[name] = count
        total += count
    result['total_trainable'] = total
    result['total_all']       = sum(p.numel() for p in model.parameters())
    return result
