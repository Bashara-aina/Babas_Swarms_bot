// ===== Stepper (Worked Example with D4 term highlight) =====
var currentStep = 0;
var totalSteps = 11;

function showStep(n) {
  currentStep = n;
  document.querySelectorAll('#mainStepper .step-content').forEach(function(el, i) {
    el.classList.toggle('active', i === n);
  });
  document.querySelectorAll('#mainStepper .step-tab').forEach(function(el, i) {
    el.classList.toggle('active', i === n);
  });
  applyTermHighlight(n);
  var stepEl = document.getElementById('step' + n);
  if (stepEl && typeof window.rerenderMath === 'function') {
    setTimeout(function() { window.rerenderMath(stepEl); }, 0);
  }
}

function nextStep() {
  if (currentStep < totalSteps - 1) showStep(currentStep + 1);
}

function prevStep() {
  if (currentStep > 0) showStep(currentStep - 1);
}

document.addEventListener('keydown', function(e) {
  if (!e.key || (e.key !== 'ArrowLeft' && e.key !== 'ArrowRight')) return;
  var t = e.target;
  if (t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.tagName === 'SELECT' || t.isContentEditable)) return;
  var stepper = document.getElementById('mainStepper');
  if (!stepper) return;
  var r = stepper.getBoundingClientRect();
  if (r.bottom < 40 || r.top > window.innerHeight - 40) return;
  e.preventDefault();
  if (e.key === 'ArrowRight') nextStep();
  else prevStep();
});

function applyTermHighlight(stepIndex) {
  document.querySelectorAll('.term-highlight').forEach(function(el) {
    el.classList.remove('term-highlight');
  });
  document.querySelectorAll('.step-keyterm').forEach(function(el) {
    el.classList.remove('term-active');
  });
  var activeStep = document.getElementById('step' + stepIndex);
  if (!activeStep) return;
  var focusEl = activeStep.querySelector('.step-keyterm');
  if (focusEl) focusEl.classList.add('term-active');
  var katexEls = activeStep.querySelectorAll('.katex');
  if (katexEls.length > 0) {
    katexEls[0].classList.add('term-highlight');
  }
}

function initLazyCharts() {
  var CHART_INIT = {
    lossChart: initLossChart,
    classDistChart: initClassDistChart,
    crossoverChart: initCrossoverChart,
    focalGammaChart: initFocalGammaChart,
    kendallSigmaChart: initKendallSigmaChart,
    wingGradientChart: initWingGradientChart
  };
  Object.keys(CHART_INIT).forEach(function(id) {
    var el = document.getElementById(id);
    if (!el) return;
    el.dataset.chart = id;
  });
  var observer = new IntersectionObserver(function(entries) {
    entries.forEach(function(entry) {
      if (!entry.isIntersecting) return;
      var canvas = entry.target;
      var key = canvas.dataset.chart;
      if (!key || canvas.dataset.chartInit === 'true') return;
      canvas.dataset.chartInit = 'true';
      var fn = CHART_INIT[key];
      if (typeof fn === 'function') fn();
      observer.unobserve(canvas);
    });
  }, { threshold: 0.1 });
  document.querySelectorAll('canvas[data-chart]').forEach(function(c) { observer.observe(c); });
}

  function animateMetric(el, target, suffix, dur) {
    if (!el) return;
    if (REDUCE_MOTION) { el.textContent = target + (suffix || ''); return; }
    dur = dur || 900;
    var start = 0;
    var startT = null;
    var tgt = parseFloat(target);
    function frame(now) {
      if (!startT) startT = now;
      var p = Math.min(1, (now - startT) / dur);
      var eased = 1 - Math.pow(1 - p, 3);
      var v = start + (tgt - start) * eased;
      el.textContent = v.toFixed(2) + (suffix || '');
      if (p < 1) requestAnimationFrame(frame);
      else el.textContent = target + (suffix || '');
    }
    requestAnimationFrame(frame);
  }

  function hexToRgb(hex) {
    var n = hex.replace('#','');
    return { r: parseInt(n.slice(0,2),16), g: parseInt(n.slice(2,4),16), b: parseInt(n.slice(4,6),16) };
  }

  var C5_ACTIVATIONS = [
    [0.10, 0.10, 0.20, 0.20, 0.30, 0.10, 0.10],

  function getC1Activation(row, col, rows, cols) {
    var x = col / cols * 480, y = row / rows * 270;
    var nearTorso  = Math.abs(x-240) < 50 && y > 70  && y < 155;
    var nearArms   = (Math.abs(x-170) < 15 && y > 75  && y < 155) ||
                     (Math.abs(x-312) < 15 && y > 75  && y < 155);
    var nearLegs   = (Math.abs(x-195) < 12 && y > 145 && y < 245) ||
                     (Math.abs(x-282) < 12 && y > 145 && y < 245);
    var nearShoulders = Math.abs(y-80) < 8 && x > 170 && x < 310;
    var nearHips   = Math.abs(y-145) < 8 && x > 190 && x < 290;
    if(nearShoulders || nearHips) return 0.45 + Math.sin(x*0.1)*0.15;
    if(nearTorso || nearArms || nearLegs) return 0.25 + Math.sin(x*0.2+y*0.3)*0.18;
    return 0.01;
  }

  function getC2Activation(row, col, rows, cols) {
    var x = col / cols * 480, y = row / rows * 270;
    var inBody = x>=170 && x<=320 && y>=50 && y<=245;
    var inArms = (x>=140 && x<=180 && y>=75 && y<=160) || (x>=300 && x<=340 && y>=75 && y<=160);
    if(inArms) return 0.18 + Math.sin(row*0.5+col*0.3)*0.12;
    if(inBody) return 0.12 + Math.sin(row*0.3+col*0.2)*0.1;
    return 0.02;
  }

  function getC3Activation(row, col, rows, cols) {
    var nx = col / cols, ny = row / rows;
    var inTorso = nx >= 0.28 && nx <= 0.72 && ny >= 0.14 && ny <= 0.48;
    var inLeftArm = nx >= 0.05 && nx <= 0.22 && ny >= 0.14 && ny <= 0.42;
    var inRightArm = nx >= 0.78 && nx <= 0.95 && ny >= 0.14 && ny <= 0.42;
    var inHead = nx >= 0.35 && nx <= 0.65 && ny >= 0 && ny <= 0.18;
    if (inRightArm) return 0.5 + Math.sin(row * 1.1 + col * 0.7) * 0.15;
    if (inTorso) return 0.3 + Math.sin(row * 0.9 + col * 0.5) * 0.1;
    if (inLeftArm) return 0.2 + Math.sin(row * 0.8 + col * 0.6) * 0.08;
    if (inHead) return 0.25 + Math.sin(row * 1.2) * 0.1;
    return 0.02;
  }

  function getC4Activation(row, col, rows, cols) {
    var cx = cols / 2, cy = rows * 0.5;
    var colf = col, rowf = row;
    var dist = Math.sqrt(Math.pow((colf - cx) / (cols / 6), 2) + Math.pow((rowf - cy) / (rows / 6), 2));
    var base = Math.max(0, 1 - dist / 2.5) * 0.6;
    var armBoost = (colf > cols * 0.45 && rowf > rows * 0.2 && rowf < rows * 0.55) ? 0.3 : 0;
    return Math.min(1, base + armBoost);
  }

  /* 7×7 reference pattern upsampled to POPW C5 grid (20×15) for canvas visualization */
  function getC5Activation(row, col, rows, cols) {
    var sy = Math.min(6, Math.floor(((row + 0.5) / rows) * 7));
    var sx = Math.min(6, Math.floor(((col + 0.5) / cols) * 7));
    return C5_ACTIVATIONS[sy][sx];
  }

  function drawFeatureMap(canvas, mode, color) {
    if (!canvas) return;
    var ctx = canvas.getContext('2d');
    var W = canvas.width, H = canvas.height;
    ctx.clearRect(0, 0, W, H);
    var rgb = hexToRgb(color);

    var configs = {
      c1: { cols: 160, rows: 120, fn: getC1Activation },
      c2: { cols: 80, rows: 60, fn: getC2Activation },
      c3: { cols: 40, rows: 30, fn: getC3Activation },
      c4: { cols: 20, rows: 15, fn: getC4Activation },
      c5: { cols: 20, rows: 15, fn: getC5Activation }
    };
    var cfg = configs[mode];
    if (!cfg) return;

    var cellW = W / cfg.cols;
    var cellH = H / cfg.rows;

    /* Always draw faint skeleton under feature activations for spatial grounding */
    var edges = [[5,6],[5,7],[7,9],[6,8],[8,10],[5,11],[6,12],[11,12],[11,13],[13,15],[12,14],[14,16],[0,5],[0,6]];
    ctx.save();
    ctx.strokeStyle = 'rgba('+rgb.r+','+rgb.g+','+rgb.b+',0.22)';
    ctx.lineWidth = 1.15;
    edges.forEach(function(e) {
      var a = JOINTS[e[0]], b = JOINTS[e[1]];
      ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y); ctx.stroke();
    });
    JOINTS.forEach(function(j) {
      ctx.beginPath();
      ctx.arc(j.x, j.y, j.wrist ? 3.2 : 2.2, 0, Math.PI * 2);
      ctx.fillStyle = 'rgba('+rgb.r+','+rgb.g+','+rgb.b+',0.20)';
      ctx.fill();
    });
    ctx.restore();

    for (var row = 0; row < cfg.rows; row++) {
      for (var col = 0; col < cfg.cols; col++) {
        var activation = cfg.fn(row, col, cfg.rows, cfg.cols);
        if (activation > 0.05) {
          ctx.fillStyle = 'rgba('+rgb.r+','+rgb.g+','+rgb.b+','+(activation * 0.7)+')';
          ctx.fillRect(col*cellW, row*cellH, cellW - 0.5, cellH - 0.5);
        }
      }
    }

    /* Grid lines for all modes; decimate for very dense maps for readability */
    {
      var denseStep = cfg.cols >= 120 ? 8 : (cfg.cols >= 50 ? 4 : (cfg.cols >= 28 ? 2 : 1));
      ctx.strokeStyle = 'rgba('+rgb.r+','+rgb.g+','+rgb.b+',' + (cfg.cols <= 28 ? '0.15' : '0.10') + ')';
      ctx.lineWidth = 0.5;
      for (var c = denseStep; c < cfg.cols; c += denseStep) {
        ctx.beginPath(); ctx.moveTo(c*cellW, 0); ctx.lineTo(c*cellW, H); ctx.stroke();
      }
      for (var r = denseStep; r < cfg.rows; r += denseStep) {
        ctx.beginPath(); ctx.moveTo(0, r*cellH); ctx.lineTo(W, r*cellH); ctx.stroke();
      }
    }

    /* C5: label high-activation cells (20×15 POPW stride-32 grid) */
    if (mode === 'c5') {
      ctx.font = 'bold 8px monospace';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      for (var sy = 0; sy < cfg.rows; sy += 2) {
        for (var sx = 0; sx < cfg.cols; sx += 2) {
          var a = getC5Activation(sy, sx, cfg.rows, cfg.cols);
          if (a > 0.6) {
            ctx.fillStyle = 'rgba(255,255,255,0.85)';
            ctx.fillText(a.toFixed(2), sx * cellW + cellW / 2, sy * cellH + cellH / 2);
          }
        }
      }
      ctx.fillStyle = 'rgba(56,189,248,0.9)';
      ctx.font = '8px monospace';
      ctx.textAlign = 'left';
      ctx.fillText('\u2190 wrist region', 13 * cellW + cellW * 0.5 - 4, 9 * cellH + 12);
      ctx.fillStyle = 'rgba(56,189,248,0.92)';
      ctx.font = '9px monospace';
      ctx.fillText('20×15 spatial (640×480 @ stride 32) — C5 [B,2048,20,15] per model.py', 8, H - 8);
    }
  }

  function fillFspGrid(withoutEl, withEl) {
    if (!withoutEl || !withEl) return;
    var WGRID = 20, HGRID = 15, N = WGRID * HGRID;
    function cell(val) {
      var d = document.createElement('div');
      d.className = 'fsp-cell';
      d.style.background = 'rgba(56,189,248,' + (0.12 + val * 0.5) + ')';
      return d;
    }
    withoutEl.innerHTML = '';
    withEl.innerHTML = '';
    for (var i = 0; i < N; i++) {
      withoutEl.appendChild(cell(0.35 + (i % 5) * 0.02));
    }
    for (var j = 0; j < N; j++) {
      var row = Math.floor(j / WGRID), col = j % WGRID;
      var r7 = Math.min(6, Math.floor(row / HGRID * 7));
      var c7 = Math.min(6, Math.floor(col / WGRID * 7));
      var v = 0.28;
      if (r7 >= 2 && r7 <= 4 && c7 >= 3 && c7 <= 5) v = 0.75 + (r7 + c7) * 0.02;
      if (r7 === 4 && c7 === 4) v = 0.95;
      withEl.appendChild(cell(v));
    }
  }

      drawFeatureMap(feat, 'c5', FEATURE_MAPS.c5.color);
      drawFilmOverlay(film, frame);
    } else if (mode === 'loss') {
      if (ml) ml.textContent = 'Wing loss';
      if (tl) tl.textContent = 'keypoint regression';
      if (al) al.textContent = '';
      drawLossAnnotations(frame);
    }

    var wrap = frame.closest('.living-frame-wrap');
    if (wrap) {
      updateDescription(wrap, mode);
      toggleFsp(wrap, mode, true);
    }

    var hint = document.getElementById('lf-arch-c5-hint');
    if (hint) {
      hint.hidden = !(frame.getAttribute('data-frame-id') === 'frame-arch' && mode === 'c5');
    }
  }

  /* ---- FIX 1: Scope joint queries to frame via data-joint-id ---- */
  function initJointTooltips(frame) {
    var fid = frame.getAttribute('data-frame-id');
    var tooltip = document.getElementById('lf-tooltip-' + fid);
    if (!tooltip) return;
    JOINTS.forEach(function(joint) {
      var el = frame.querySelector('[data-joint-id="' + joint.id + '"]');
      if (!el) return;
      el.addEventListener('mouseenter', function() {
        var c = joint.coords;
        var html = '<span style="color:#fbbf24">' + joint.id + '</span> <strong>' + joint.name + '</strong><br>' +
          '<span style="color:#38bdf8">x=' + c[0].toFixed(2) + ', y=' + c[1].toFixed(2) + ', z=' + c[2].toFixed(2) + '</span>';
        if (joint.wrist) html += '<br><span style="color:#4ade80">★ Discriminative for assembly</span>';
        if (joint.root) html += '<br><span style="color:#fbbf24">★ Coordinate origin</span>';
        if (joint.key) html += '<br><span style="color:#7dd3fc">★ Shoulder pivot</span>';
        tooltip.innerHTML = html;
        var fr = frame.getBoundingClientRect();
        var jr = el.getBoundingClientRect();
        var tx = jr.left - fr.left + jr.width / 2;
        var ty = jr.top - fr.top - 8;
        tooltip.style.left = Math.min(Math.max(4, tx - 80), fr.width - 200) + 'px';
        tooltip.style.top = Math.max(4, ty - 72) + 'px';
        tooltip.classList.add('visible');
      });
      el.addEventListener('mouseleave', function() {
        tooltip.classList.remove('visible');
      });
    });
  }

  function initModeButtons(frame) {
    var fid = frame.getAttribute('data-frame-id');
    if (!fid) return;
    document.querySelectorAll('[data-frame="' + fid + '"].lf-btn').forEach(function(btn) {
      var modeName = btn.getAttribute('data-mode') || 'unknown';
      if (!btn.getAttribute('aria-label')) {
        btn.setAttribute('aria-label', 'Set ' + fid + ' frame to ' + modeName + ' mode');
      }
      btn.addEventListener('click', function() {
        var mode = btn.getAttribute('data-mode');
        document.querySelectorAll('[data-frame="' + fid + '"].lf-btn').forEach(function(b) {
          b.classList.toggle('active', b === btn);
        });
        frame.dataset.mode = mode;
        renderCurrentMode(frame);
      });
    });
  }

  function bindSliders() {
    document.querySelectorAll('.lf-gamma-slider').forEach(function(sl) {
      sl.addEventListener('input', function() {
        var id = sl.getAttribute('data-lf-target');
        var fr = document.querySelector('[data-frame-id="' + id + '"]');
        if (fr) {
          fr.dataset.gamma = sl.value;
          var d = document.querySelector('.lf-gamma-disp[data-lf-target="' + id + '"]');
          if (d) d.textContent = parseFloat(sl.value).toFixed(2);
          fr.dataset.mode = 'film';
          document.querySelectorAll('[data-frame="' + id + '"].lf-btn').forEach(function(b) {
            b.classList.toggle('active', b.getAttribute('data-mode') === 'film');
          });
          renderCurrentMode(fr);
        }
      });
    });
    document.querySelectorAll('.lf-beta-slider').forEach(function(sl) {
      sl.addEventListener('input', function() {
        var id = sl.getAttribute('data-lf-target');
        var fr = document.querySelector('[data-frame-id="' + id + '"]');
        if (fr) {
          fr.dataset.beta = sl.value;
          var d = document.querySelector('.lf-beta-disp[data-lf-target="' + id + '"]');
          if (d) d.textContent = parseFloat(sl.value).toFixed(2);
          fr.dataset.mode = 'film';
          document.querySelectorAll('[data-frame="' + id + '"].lf-btn').forEach(function(b) {
            b.classList.toggle('active', b.getAttribute('data-mode') === 'film');
          });
          renderCurrentMode(fr);
        }
      });
    });
  }

  /* ---- FIX 2: Corrected step-to-mode mapping + stepper-change event ---- */
  var STEP_MODES = ['rgb','skeleton','skeleton','film','loss','c5','film','loss','film','film','film'];

  function syncWorkedFrame(step) {
    var frame = document.querySelector('[data-frame-id="frame-worked"]');
    if (!frame) return;
    var mode = STEP_MODES[step] || 'rgb';
    frame.dataset.mode = mode;
    /* Clear annotations on backward navigation */
    var ann = frame.querySelector('.lf-annotations');
    if (ann) while (ann.firstChild) ann.removeChild(ann.firstChild);
    document.querySelectorAll('[data-frame="frame-worked"].lf-btn').forEach(function(b) {
      b.classList.toggle('active', b.getAttribute('data-mode') === mode);
    });
    renderCurrentMode(frame);
  }

  function patchStepper() {
    if (window.__popwLivingFrameStepperPatched) return;
    var orig = window.showStep;
    if (typeof orig !== 'function') return;
    window.__popwLivingFrameStepperPatched = true;
    window.showStep = function(n) {
      orig(n);
      syncWorkedFrame(n);
      try {
        document.dispatchEvent(new CustomEvent('stepper-change', { bubbles: true, detail: { step: n } }));
      } catch (e) {}
    };
    syncWorkedFrame(typeof window.currentStep === 'number' ? window.currentStep : 0);
  }

  /* FIX 2: Listen for stepper-change on any frame */
  document.addEventListener('stepper-change', function(e) {
    var frame = document.querySelector('[data-frame-id="frame-worked"]');
    if (!frame) return;
    var mode = STEP_MODES[e.detail.step] || 'rgb';
    frame.dataset.mode = mode;
    renderCurrentMode(frame);
    document.querySelectorAll('[data-frame="frame-worked"].lf-btn').forEach(function(b) {
      b.classList.toggle('active', b.getAttribute('data-mode') === mode);
    });
    var stepEl = document.getElementById('step' + e.detail.step);
    if (stepEl && typeof window.rerenderMath === 'function') {
      setTimeout(function() { window.rerenderMath(stepEl); }, 0);
    }
  });

  function initAll() {
    document.querySelectorAll('.living-frame').forEach(function(frame) {
      if (frame.dataset.lfBound) {
        renderCurrentMode(frame);
        return;
      }
      frame.dataset.lfBound = '1';
      initJointTooltips(frame);
      initModeButtons(frame);
      renderCurrentMode(frame);
    });
    if (!window.__lfSlidersBound) {
      window.__lfSlidersBound = true;
      bindSliders();
    }
    patchStepper();
  }

  function rerenderAll() {
    document.querySelectorAll('.living-frame').forEach(function(frame) { renderCurrentMode(frame); });
  }

  window.LivingFrame = { initAll: initAll, renderCurrentMode: renderCurrentMode, syncWorkedFrame: syncWorkedFrame, rerenderAll: rerenderAll };

  try {
    var mo = new MutationObserver(function() { rerenderAll(); });
    mo.observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] });
  } catch (e) {}

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initAll);
  } else {
    initAll();
  }
  window.addEventListener('load', function() {
    try {
      initAll();
    } catch (e) {}
  });

  (function wireArchDiagram() {
    var wrap = document.querySelector('.arch-flow-wrap');
    if (!wrap) return;
    function setArchMode(mode) {
      if (!mode) return;
      var frame = document.querySelector('[data-frame-id="frame-arch"]');
      if (!frame) return;
      frame.dataset.mode = mode;
      document.querySelectorAll('[data-frame="frame-arch"].lf-btn').forEach(function(b) {
        b.classList.toggle('active', b.getAttribute('data-mode') === mode);
      });
      if (window.LivingFrame && typeof window.LivingFrame.renderCurrentMode === 'function') {
        window.LivingFrame.renderCurrentMode(frame);
      }
      try { frame.scrollIntoView({ behavior: 'smooth', block: 'nearest' }); } catch (e) {}
    }
    wrap.addEventListener('click', function(ev) {
      var hit = ev.target.closest('.arch-hit[data-arch-mode]');
      if (!hit) return;
      ev.preventDefault();
      setArchMode(hit.getAttribute('data-arch-mode'));
    });
    wrap.addEventListener('keydown', function(ev) {
      if (ev.key !== 'Enter' && ev.key !== ' ') return;
      var hit = ev.target.closest('.arch-hit[data-arch-mode]');
      if (!hit) return;
      ev.preventDefault();
      setArchMode(hit.getAttribute('data-arch-mode'));
    });
  })();
})();
