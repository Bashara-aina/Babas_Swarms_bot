"""
Sequential Pipeline Baseline (P2.2)
=====================================
Measures the FPS, GFLOPs, and memory of running 3 separate models
sequentially to solve the same 3 tasks as WorkerNet:
  1. YOLOv8n  for object detection
  2. ViTPose-S (or OpenPose fallback) for pose estimation
  3. ResNet-50 single-frame for activity classification

This establishes the "traditional sequential pipeline" baseline
that WorkerNet replaces with a single forward pass.

Usage:
    python sequential_baseline.py
    python sequential_baseline.py --num_images 100

Requirements:
    pip install ultralytics

Author: Bashara
Date: February 2026
"""

import argparse
import logging
import time
from pathlib import Path

import numpy as np
import torch
import torch.nn as nn
from torchvision.models import resnet50, ResNet50_Weights

import config as C

logger = logging.getLogger(__name__)


# =============================================================================
# Individual Task Models
# =============================================================================
class StandaloneActivityClassifier(nn.Module):
    """ResNet-50 single-frame activity classifier (same arch as WorkerNet head)."""

    def __init__(self, num_classes=C.NUM_ACT_CLASSES):
        super().__init__()
        backbone = resnet50(weights=ResNet50_Weights.IMAGENET1K_V2)
        # Remove final FC
        self.features = nn.Sequential(
            backbone.conv1, backbone.bn1, backbone.relu, backbone.maxpool,
            backbone.layer1, backbone.layer2, backbone.layer3, backbone.layer4,
        )
        self.pool = nn.AdaptiveAvgPool2d(1)
        self.fc = nn.Sequential(
            nn.Flatten(),
            nn.Linear(2048, 512), nn.ReLU(True),
            nn.Dropout(0.3),
            nn.Linear(512, num_classes),
        )

    def forward(self, x):
        feat = self.features(x)
        return self.fc(self.pool(feat))


def load_yolo_detector(device):
    """Load YOLOv8n for detection. Returns model or None if unavailable."""
    try:
        from ultralytics import YOLO
        model = YOLO('yolov8n.pt')
        # Warmup
        dummy = torch.randn(1, 3, C.IMG_HEIGHT, C.IMG_WIDTH)
        _ = model.predict(dummy.permute(0, 2, 3, 1).numpy()[0], verbose=False)
        return model
    except ImportError:
        logger.warning('ultralytics not installed. pip install ultralytics')
        return None
    except Exception as e:
        logger.warning(f'Failed to load YOLOv8n: {e}')
        return None


def load_pose_model(device):
    """
    Load a standalone pose estimation model.
    Uses a simple heatmap model as proxy (same architecture as WorkerNet pose head
    but with its own backbone). For a fair comparison, this represents the
    minimum viable pose model in a sequential pipeline.
    """
    class StandalonePoseModel(nn.Module):
        def __init__(self):
            super().__init__()
            backbone = resnet50(weights=ResNet50_Weights.IMAGENET1K_V2)
            self.features = nn.Sequential(
                backbone.conv1, backbone.bn1, backbone.relu, backbone.maxpool,
                backbone.layer1, backbone.layer2,
            )
            self.deconv = nn.Sequential(
                nn.ConvTranspose2d(512, 256, 4, stride=2, padding=1),
                nn.BatchNorm2d(256), nn.ReLU(True),
                nn.ConvTranspose2d(256, 256, 4, stride=2, padding=1),
                nn.BatchNorm2d(256), nn.ReLU(True),
                nn.ConvTranspose2d(256, C.NUM_KEYPOINTS, 4, stride=2, padding=1),
            )

        def forward(self, x):
            feat = self.features(x)
            return self.deconv(feat)

    model = StandalonePoseModel().to(device).eval()
    return model


# =============================================================================
# Benchmarking
# =============================================================================
def benchmark_sequential(device, num_images=100, warmup=20):
    """
    Benchmark the sequential pipeline: det -> pose -> act.
    Uses PyTorch models directly (no actual image loading needed).
    """
    logger.info('Loading sequential pipeline models ...')

    # 1. Detection: YOLOv8n (or ResNet-50-based proxy if YOLO unavailable)
    yolo = load_yolo_detector(device)
    use_yolo = yolo is not None

    if not use_yolo:
        logger.info('  Using ResNet-50 proxy for detection (YOLO unavailable)')

        class DetProxy(nn.Module):
            def __init__(self):
                super().__init__()
                backbone = resnet50(weights=ResNet50_Weights.IMAGENET1K_V2)
                self.features = nn.Sequential(
                    backbone.conv1, backbone.bn1, backbone.relu, backbone.maxpool,
                    backbone.layer1, backbone.layer2, backbone.layer3, backbone.layer4,
                )
                self.pool = nn.AdaptiveAvgPool2d(1)
                self.fc = nn.Linear(2048, C.NUM_DET_CLASSES * 5)

            def forward(self, x):
                return self.fc(self.pool(self.features(x)).flatten(1))

        det_model = DetProxy().to(device).eval()
    else:
        det_model = None

    # 2. Pose: Standalone heatmap model
    pose_model = load_pose_model(device)
    logger.info('  Pose model loaded')

    # 3. Activity: Standalone ResNet-50
    act_model = StandaloneActivityClassifier().to(device).eval()
    logger.info('  Activity model loaded')

    dummy = torch.randn(1, 3, C.IMG_HEIGHT, C.IMG_WIDTH, device=device)

    # Warmup
    logger.info(f'  Warming up ({warmup} iterations) ...')
    with torch.no_grad():
        for _ in range(warmup):
            if use_yolo:
                _ = yolo.predict(
                    (dummy[0].permute(1, 2, 0).cpu().numpy() * 255).astype(np.uint8),
                    verbose=False,
                )
            else:
                _ = det_model(dummy)
            _ = pose_model(dummy)
            _ = act_model(dummy)
    torch.cuda.synchronize()

    # Benchmark: measure each model separately + total
    logger.info(f'  Benchmarking ({num_images} iterations) ...')
    torch.cuda.reset_peak_memory_stats(device)

    det_times, pose_times, act_times, total_times = [], [], [], []
    img_np = (dummy[0].permute(1, 2, 0).cpu().numpy() * 255).astype(np.uint8)

    with torch.no_grad():
        for _ in range(num_images):
            torch.cuda.synchronize()
            t_total_start = time.perf_counter()

            # Step 1: Detection
            torch.cuda.synchronize()
            t0 = time.perf_counter()
            if use_yolo:
                _ = yolo.predict(img_np, verbose=False)
            else:
                _ = det_model(dummy)
            torch.cuda.synchronize()
            det_times.append(time.perf_counter() - t0)

            # Step 2: Pose
            torch.cuda.synchronize()
            t0 = time.perf_counter()
            _ = pose_model(dummy)
            torch.cuda.synchronize()
            pose_times.append(time.perf_counter() - t0)

            # Step 3: Activity
            torch.cuda.synchronize()
            t0 = time.perf_counter()
            _ = act_model(dummy)
            torch.cuda.synchronize()
            act_times.append(time.perf_counter() - t0)

            total_times.append(time.perf_counter() - t_total_start)

    peak_mem = torch.cuda.max_memory_allocated(device) / 1e6

    det_med = np.median(det_times)
    pose_med = np.median(pose_times)
    act_med = np.median(act_times)
    total_med = np.median(total_times)

    results = {
        'det_latency_ms': det_med * 1000,
        'pose_latency_ms': pose_med * 1000,
        'act_latency_ms': act_med * 1000,
        'total_latency_ms': total_med * 1000,
        'total_fps': 1.0 / total_med,
        'peak_gpu_mb': peak_mem,
        'det_model': 'YOLOv8n' if use_yolo else 'ResNet-50-proxy',
        'pose_model': 'ResNet-50-heatmap',
        'act_model': 'ResNet-50-FC',
    }

    # GFLOPs measurement
    try:
        from fvcore.nn import FlopCountAnalysis
        gflops_total = 0.0

        if not use_yolo:
            flops = FlopCountAnalysis(det_model, dummy)
            g = flops.total() / 1e9
            results['det_gflops'] = g
            gflops_total += g

        flops = FlopCountAnalysis(pose_model, dummy)
        g = flops.total() / 1e9
        results['pose_gflops'] = g
        gflops_total += g

        flops = FlopCountAnalysis(act_model, dummy)
        g = flops.total() / 1e9
        results['act_gflops'] = g
        gflops_total += g

        if use_yolo:
            # Approximate YOLOv8n at ~8.7 GFLOPs (published)
            results['det_gflops'] = 8.7
            gflops_total += 8.7

        results['total_gflops'] = gflops_total
    except ImportError:
        logger.warning('fvcore not installed, skipping GFLOPs')
        results['total_gflops'] = None

    # Parameter counts
    param_count = 0
    if not use_yolo:
        param_count += sum(p.numel() for p in det_model.parameters())
    else:
        param_count += 3.2e6  # YOLOv8n published param count
    param_count += sum(p.numel() for p in pose_model.parameters())
    param_count += sum(p.numel() for p in act_model.parameters())
    results['total_params'] = int(param_count)

    return results


def benchmark_workernet(device, num_images=100, warmup=20, checkpoint=None):
    """Benchmark WorkerNet for direct comparison."""
    from model import MultiTaskIKEA

    if checkpoint and Path(checkpoint).exists():
        model = MultiTaskIKEA(pretrained=False, use_film=C.USE_FILM).to(device)
        ckpt = torch.load(checkpoint, map_location=device)
        model.load_state_dict(ckpt['model'], strict=False)
    else:
        model = MultiTaskIKEA(pretrained=True, use_film=C.USE_FILM).to(device)

    model.eval()
    dummy = torch.randn(1, 3, C.IMG_HEIGHT, C.IMG_WIDTH, device=device)

    with torch.no_grad():
        for _ in range(warmup):
            _ = model(dummy)
    torch.cuda.synchronize()

    torch.cuda.reset_peak_memory_stats(device)
    times = []
    with torch.no_grad():
        for _ in range(num_images):
            torch.cuda.synchronize()
            t0 = time.perf_counter()
            _ = model(dummy)
            torch.cuda.synchronize()
            times.append(time.perf_counter() - t0)

    peak_mem = torch.cuda.max_memory_allocated(device) / 1e6
    total_med = np.median(times)

    results = {
        'total_latency_ms': total_med * 1000,
        'total_fps': 1.0 / total_med,
        'peak_gpu_mb': peak_mem,
        'total_params': sum(p.numel() for p in model.parameters()),
    }

    try:
        from fvcore.nn import FlopCountAnalysis
        flops = FlopCountAnalysis(model, dummy)
        results['total_gflops'] = flops.total() / 1e9
    except ImportError:
        results['total_gflops'] = None

    return results

# CLI
# =============================================================================
def main(args):
    logging.basicConfig(
        level=logging.INFO,
        format='%(asctime)s | %(levelname)s | %(message)s',
    )

    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    if device.type != 'cuda':
        logger.error('CUDA required for benchmark.')
        return

    logger.info(f'GPU: {torch.cuda.get_device_name()}')
    logger.info(f'Input: {C.IMG_HEIGHT}x{C.IMG_WIDTH}')

    # Benchmark sequential pipeline
    print('\n' + '=' * 60)
    print('SEQUENTIAL PIPELINE BENCHMARK')
    print('=' * 60)
    seq = benchmark_sequential(device, num_images=args.num_images)

    print(f'\n  Detection ({seq["det_model"]}):')
    print(f'    Latency: {seq["det_latency_ms"]:.1f} ms')
    if 'det_gflops' in seq:
        print(f'    GFLOPs : {seq["det_gflops"]:.2f}')

    print(f'  Pose ({seq["pose_model"]}):')
    print(f'    Latency: {seq["pose_latency_ms"]:.1f} ms')
    if 'pose_gflops' in seq:
        print(f'    GFLOPs : {seq["pose_gflops"]:.2f}')

    print(f'  Activity ({seq["act_model"]}):')
    print(f'    Latency: {seq["act_latency_ms"]:.1f} ms')
    if 'act_gflops' in seq:
        print(f'    GFLOPs : {seq["act_gflops"]:.2f}')

    print(f'\n  TOTAL Sequential:')
    print(f'    FPS         : {seq["total_fps"]:.1f}')
    print(f'    Latency     : {seq["total_latency_ms"]:.1f} ms')
    print(f'    Peak GPU    : {seq["peak_gpu_mb"]:.0f} MB')
    print(f'    Parameters  : {seq["total_params"]:,}')
    if seq.get('total_gflops'):
        print(f'    GFLOPs      : {seq["total_gflops"]:.2f}')

    # Benchmark WorkerNet
    print('\n' + '=' * 60)
    print('WORKERNET (UNIFIED) BENCHMARK')
    print('=' * 60)
    wn = benchmark_workernet(device, num_images=args.num_images,
                             checkpoint=args.checkpoint)

    print(f'\n  TOTAL WorkerNet:')
    print(f'    FPS         : {wn["total_fps"]:.1f}')
    print(f'    Latency     : {wn["total_latency_ms"]:.1f} ms')
    print(f'    Peak GPU    : {wn["peak_gpu_mb"]:.0f} MB')
    print(f'    Parameters  : {wn["total_params"]:,}')
    if wn.get('total_gflops'):
        print(f'    GFLOPs      : {wn["total_gflops"]:.2f}')

    # Comparison table
    print('\n' + '=' * 60)
    print('COMPARISON TABLE (copy to paper)')
    print('=' * 60)
    print(f'{"":20s} | {"Sequential":>12s} | {"WorkerNet":>12s} | {"Speedup":>8s}')
    print('-' * 60)
    print(f'{"Tasks":20s} | {"3 (serial)":>12s} | {"3 (unified)":>12s} | {"":>8s}')
    print(f'{"Single Forward Pass":20s} | {"No":>12s} | {"Yes":>12s} | {"":>8s}')

    speedup_fps = wn['total_fps'] / max(seq['total_fps'], 0.01)
    print(f'{"FPS":20s} | {seq["total_fps"]:>12.1f} | {wn["total_fps"]:>12.1f} | {speedup_fps:>7.1f}x')
    print(f'{"Latency (ms)":20s} | {seq["total_latency_ms"]:>12.1f} | {wn["total_latency_ms"]:>12.1f} | {"":>8s}')
    print(f'{"Peak GPU (MB)":20s} | {seq["peak_gpu_mb"]:>12.0f} | {wn["peak_gpu_mb"]:>12.0f} | {"":>8s}')
    print(f'{"Parameters":20s} | {seq["total_params"]:>12,} | {wn["total_params"]:>12,} | {"":>8s}')

    if seq.get('total_gflops') and wn.get('total_gflops'):
        ratio = seq['total_gflops'] / max(wn['total_gflops'], 0.01)
        print(f'{"GFLOPs":20s} | {seq["total_gflops"]:>12.2f} | {wn["total_gflops"]:>12.2f} | {ratio:>7.1f}x')

    print('=' * 60)


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Sequential vs. WorkerNet Benchmark')
    parser.add_argument('--num_images', type=int, default=100,
                        help='Number of images for benchmark')
    parser.add_argument('--checkpoint', type=str, default=None,
                        help='WorkerNet checkpoint path (optional)')
    main(parser.parse_args())
