import sys, os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

#!/usr/bin/env python3
"""Quick test to verify dataloader speed and correctness."""
import sys
import logging
import time
from ikea_dataset import build_dataloaders
import config as C

logging.basicConfig(level=logging.INFO)

print("\n" + "="*60)
print("DATALOADER TEST")
print("="*60)

try:
    print("\n1. Building dataloaders...")
    loaders = build_dataloaders(batch_size=C.BATCH_SIZE, num_workers=2)
    train_loader = loaders['train']
    print(f"   ✅ Train loader: {len(train_loader)} batches")

    print("\n2. Testing first 10 batches with per-batch timing...")
    batch_times = []
    t0 = time.time()
    for i, (images, targets) in enumerate(train_loader):
        t1 = time.time()
        batch_times.append(t1 - t0)
        print(
            f"   Batch {i+1:2d}/10: "
            f"load={t1-t0:.3f}s | "
            f"images={list(images.shape)} | "
            f"det={len(targets['detection'])} | "
            f"pose={list(targets['keypoints'].shape)} | "
            f"act={list(targets['activity'].shape)}"
        )
        t0 = time.time()
        if i >= 9:
            break

    # ✅ FIX: Accurate per-batch average (excludes first warm-up batch)
    avg_time = sum(batch_times[1:]) / max(len(batch_times) - 1, 1)
    print(f"\n✅ Per-batch avg (batches 2-10): {avg_time:.3f}s")
    print(f"   Estimated epoch time: {avg_time * len(train_loader) / 3600:.1f} hours")
    print(f"   DEBUG_MODE={C.DEBUG_MODE}, "
          f"BATCH_SIZE={C.BATCH_SIZE}, "
          f"NUM_WORKERS=2")

except Exception as e:
    print(f"\n❌ FAILED: {e}")
    import traceback
    traceback.print_exc()
    sys.exit(1)

print("\n" + "="*60)
print("Test complete.")
print("="*60 + "\n")
