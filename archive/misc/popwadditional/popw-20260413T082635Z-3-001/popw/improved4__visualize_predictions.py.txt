# ── COCO-17 skeleton ──────────────────────────────────────────────────────────
SKELETON = [
    (0, 1), (0, 2), (1, 3), (2, 4),          # head
    (5, 7), (7, 9),                           # left arm
    (6, 8), (8, 10),                          # right arm
    (5, 6), (5, 11), (6, 12), (11, 12),       # shoulders + torso
    (11, 13), (13, 15),                       # left leg
    (12, 14), (14, 16),                       # right leg
]
LIMB_COLORS = plt.cm.hsv(np.linspace(0, 0.9, len(SKELETON)))
KP_COLOR_GT   = (0.2, 1.0, 0.2)   # green for GT
KP_COLOR_PRED = (1.0, 0.6, 0.1)   # orange for prediction
DET_PALETTE = plt.cm.tab10(np.linspace(0, 1, max(C.NUM_DET_CLASSES + 1, 10)))

IMAGENET_MEAN_T = None   # initialized after device is known
IMAGENET_STD_T  = None


# ── Helpers: decode / NMS  (same logic as evaluate.py) ───────────────────────
def decode_boxes(anchors: np.ndarray, deltas: np.ndarray) -> np.ndarray:
    a_cx = (anchors[:, 0] + anchors[:, 2]) / 2
    a_cy = (anchors[:, 1] + anchors[:, 3]) / 2
    a_w  = anchors[:, 2] - anchors[:, 0]
    a_h  = anchors[:, 3] - anchors[:, 1]
    dx, dy = deltas[:, 0], deltas[:, 1]
    dw = np.clip(deltas[:, 2], -4, 4)
    dh = np.clip(deltas[:, 3], -4, 4)
    pw, ph = np.exp(dw) * a_w, np.exp(dh) * a_h
    cx, cy = dx * a_w + a_cx, dy * a_h + a_cy
    return np.stack([cx - pw / 2, cy - ph / 2, cx + pw / 2, cy + ph / 2], axis=1)


def nms_numpy(boxes: np.ndarray, scores: np.ndarray, iou_thresh: float = 0.5) -> np.ndarray:
    if len(boxes) == 0:
        return np.array([], dtype=np.int64)
    x1, y1, x2, y2 = boxes[:, 0], boxes[:, 1], boxes[:, 2], boxes[:, 3]
    areas = np.maximum((x2 - x1) * (y2 - y1), 0)
    order = scores.argsort()[::-1]
    keep = []
    while order.size > 0:
        i = order[0]
        keep.append(i)
        if order.size == 1:
            break
        xx1 = np.maximum(x1[i], x1[order[1:]])
        yy1 = np.maximum(y1[i], y1[order[1:]])
        xx2 = np.minimum(x2[i], x2[order[1:]])
        yy2 = np.minimum(y2[i], y2[order[1:]])
        inter = np.maximum(xx2 - xx1, 0) * np.maximum(yy2 - yy1, 0)
        iou = inter / (areas[i] + areas[order[1:]] - inter + 1e-6)
        order = order[np.where(iou <= iou_thresh)[0] + 1]
    return np.array(keep, dtype=np.int64)


def run_nms_per_class(
    cls_sig: np.ndarray,       # (N_anc, num_classes)
    reg_np: np.ndarray,         # (N_anc, 4)
    anc_np: np.ndarray,         # (N_anc, 4)
    score_thresh: float = 0.30,
    nms_thresh: float   = 0.50,
):
    max_scores = cls_sig.max(axis=1)
    keep_mask  = max_scores > score_thresh
    if keep_mask.sum() == 0:
        return (
            np.zeros((0, 4), dtype=np.float32),
            np.zeros(0, dtype=np.int64),
            np.zeros(0, dtype=np.float32),
        )
    sc_k  = max_scores[keep_mask]
    cl_k  = cls_sig[keep_mask].argmax(axis=1)
    reg_k = reg_np[keep_mask]
    anc_k = anc_np[keep_mask]
    pb_k  = decode_boxes(anc_k, reg_k)

    final_boxes, final_labels, final_scores = [], [], []
    for ci in range(C.NUM_DET_CLASSES):
        cm = cl_k == ci
        if cm.sum() == 0:
            continue
        nk = nms_numpy(pb_k[cm], sc_k[cm], nms_thresh)
        final_boxes.append(pb_k[cm][nk])
        final_labels.append(np.full(len(nk), ci, dtype=np.int64))
        final_scores.append(sc_k[cm][nk])

    if not final_boxes:
        return (
            np.zeros((0, 4), dtype=np.float32),
            np.zeros(0, dtype=np.int64),
            np.zeros(0, dtype=np.float32),
        )
    return (
        np.concatenate(final_boxes),
        np.concatenate(final_labels),
        np.concatenate(final_scores),
    )


# ── Visualization drawing ─────────────────────────────────────────────────────
def tensor_to_rgb(img_t: torch.Tensor) -> np.ndarray:
    """CHW uint8 or float → HWC float [0,1] ready to imshow."""
    arr = img_t.cpu().numpy().astype(np.float32)
    if arr.max() > 1.5:            # uint8
        arr /= 255.0
    arr = arr.transpose(1, 2, 0)   # HWC
    return np.clip(arr, 0, 1)


def draw_skeleton(
    ax: plt.Axes,
    kpts: np.ndarray,              # (17, 3)  x, y, conf
    conf_thresh: float = 0.1,
    kp_color: tuple = KP_COLOR_GT,
    linewidth: float = 2.0,
):
    for ci, (i, j) in enumerate(SKELETON):
        if kpts[i, 2] > conf_thresh and kpts[j, 2] > conf_thresh:
            ax.plot(
                [kpts[i, 0], kpts[j, 0]],
                [kpts[i, 1], kpts[j, 1]],
                color=LIMB_COLORS[ci], linewidth=linewidth, alpha=0.9,
                zorder=3,
            )
    for k in range(len(kpts)):
        if kpts[k, 2] > conf_thresh:
            ax.plot(
                kpts[k, 0], kpts[k, 1],
                'o', color=kp_color, markersize=5,
                markeredgecolor='white', markeredgewidth=0.5,
                zorder=4,
            )


def draw_det_boxes(
    ax: plt.Axes,
    boxes: np.ndarray,             # (N, 4) xyxy
    labels: np.ndarray,            # 0-indexed
    scores: np.ndarray | None = None,
):
    for bi in range(len(boxes)):
        x1, y1, x2, y2 = boxes[bi]
        lbl = int(labels[bi])
        col = DET_PALETTE[lbl % len(DET_PALETTE)]
        rect = mpatches.FancyBboxPatch(
            (x1, y1), max(x2 - x1, 1), max(y2 - y1, 1),
            boxstyle='square,pad=0', linewidth=2,
            edgecolor=col, facecolor='none', zorder=2,
        )
        ax.add_patch(rect)
        name = C.DET_CLASS_NAMES.get(lbl + 1, f'cls{lbl}')
        label_str = f'{name}' if scores is None else f'{name} {scores[bi]:.2f}'
        ax.text(
            x1 + 2, max(y1 - 4, 4), label_str,
            fontsize=7, color='white', fontweight='bold',
            bbox=dict(facecolor=col, alpha=0.80, pad=1.5, edgecolor='none'),
            zorder=5,
        )


def style_ax(ax: plt.Axes, title: str, color: str = 'white'):
    ax.axis('off')
    ax.set_title(title, fontsize=8, color=color, pad=5, fontweight='bold',
                 loc='center', wrap=True)


# ── Inference helpers ─────────────────────────────────────────────────────────
def normalize(img_t: torch.Tensor, device: torch.device) -> torch.Tensor:
    """uint8 CHW tensor → float normalized CHW tensor on device."""
    x = img_t.float().to(device) / 255.0
    x = (x - IMAGENET_MEAN_T) / IMAGENET_STD_T
    return x.unsqueeze(0)   # 1CHW


@torch.no_grad()
def infer(model: MultiTaskIKEA, img_t: torch.Tensor, device: torch.device):
    x = normalize(img_t, device)
    out = model(x)
    return {
        'act_pred':  int(out['act_logits'][0].argmax().item()),
        'cls_sig':   torch.sigmoid(out['cls_preds'][0]).cpu().numpy(),   # (N_anc, 7)
        'reg_np':    out['reg_preds'][0].cpu().numpy(),                  # (N_anc, 4)
        'anc_np':    out['anchors'].cpu().numpy(),                       # (N_anc, 4) — not batched
        'kp_np':     out['keypoints'][0].cpu().numpy(),                  # (17, 2)
    }


# ── Main ─────────────────────────────────────────────────────────────────────
def main() -> None:
    global IMAGENET_MEAN_T, IMAGENET_STD_T

    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    print(f'Device: {device}')

    IMAGENET_MEAN_T = torch.tensor([0.485, 0.456, 0.406], device=device).view(3, 1, 1)
    IMAGENET_STD_T  = torch.tensor([0.229, 0.224, 0.225], device=device).view(3, 1, 1)

    # ── Load model ────────────────────────────────────────────────────────────
    ckpt_path = C.CHECKPOINT_DIR / 'best.pth'
    model = MultiTaskIKEA(
        pretrained=False,
        use_film=C.USE_FILM,
    ).to(device)
    ckpt  = torch.load(ckpt_path, map_location=device, weights_only=False)
    state = ckpt.get('model', ckpt.get('model_state_dict', ckpt))
    model.load_state_dict(state)
    model.eval()
    print(f'Loaded: {ckpt_path}')

    # ── Load val dataset ──────────────────────────────────────────────────────
    val_ds = IKEAMultiTaskDataset(split='val', augment=False)
    N = len(val_ds)
    print(f'Val samples: {N}')

    # ── Pass 1: collect activity predictions for per-class accuracy ───────────
    print('Pass 1/2: collecting activity predictions …')
    class_correct: dict = defaultdict(int)
    class_total:   dict = defaultdict(int)
    lightweight: list   = []   # (idx, gt_act, pred_act, has_gt_boxes)

    for idx in range(N):
        img_t, targets = val_ds[idx]
        gt_act    = int(targets['activity'].item())
        has_boxes = targets['boxes'].shape[0] > 0

        out = infer(model, img_t, device)
        pred_act = out['act_pred']

        class_total[gt_act]   += 1
        class_correct[gt_act] += int(pred_act == gt_act)
        lightweight.append((idx, gt_act, pred_act, has_boxes))

        if (idx + 1) % 200 == 0:
            print(f'  {idx+1}/{N}')

    per_class_acc = {
        c: (class_correct[c] / class_total[c] if class_total[c] > 0 else 0.0)
        for c in class_total
    }
    overall_acc = sum(class_correct.values()) / max(sum(class_total.values()), 1)
    print(f'Overall val accuracy: {overall_acc * 100:.1f}%')

    # ── Pass 2: select 10 samples from 10 distinct classes w/ boxes ───────────
    # Priority order: (1) has_gt_boxes, (2) correctly predicted, (3) class freq
    best_for_class: dict = {}   # cls_id → (idx, pred_act)
    for idx, gt_act, pred_act, has_boxes in lightweight:
        if not has_boxes:
            continue
        if gt_act not in best_for_class:
            best_for_class[gt_act] = (idx, pred_act)
        else:
            # prefer correctly-predicted sample
            cur_idx, cur_pred = best_for_class[gt_act]
            if pred_act == gt_act and cur_pred != gt_act:
                best_for_class[gt_act] = (idx, pred_act)

    # Take top-10 by class frequency (most-represented classes look best)
    sorted_classes = sorted(
        best_for_class.keys(),
        key=lambda c: -class_total[c],
    )[:10]
    print(f'\nSelected classes ({len(sorted_classes)}):')
    for c in sorted_classes:
        print(f'  [{c:02d}] {C.ACT_CLASS_NAMES[c]:40s}  total={class_total[c]}  '
              f'acc={per_class_acc[c]*100:.1f}%')

    # ── Pass 3: re-run inference on chosen samples, render figures ─────────────
    print('\nPass 2/2: rendering visualizations …')

    fig, axes = plt.subplots(10, 2, figsize=(22, 55), dpi=120)
    fig.patch.set_facecolor('#0d1117')
    plt.subplots_adjust(hspace=0.25, wspace=0.04)

    SCORE_THRESH = 0.30
    NMS_THRESH   = 0.50

    for row_i, act_cls in enumerate(sorted_classes):
        sample_idx, _ = best_for_class[act_cls]
        img_t, targets = val_ds[sample_idx]

        gt_boxes  = targets['boxes'].numpy()        # (N_gt, 4) xyxy
        gt_labels = targets['labels'].numpy()        # 0-indexed
        gt_kpts   = targets['keypoints'].numpy()     # (17, 3)

        out = infer(model, img_t, device)

        pred_boxes, pred_labels, pred_scores = run_nms_per_class(
            out['cls_sig'], out['reg_np'], out['anc_np'],
            score_thresh=SCORE_THRESH, nms_thresh=NMS_THRESH,
        )

        # Predicted keypoints: coords from heatmap argmax, show all with conf=1
        pred_kpts_viz = np.ones((C.NUM_KEYPOINTS, 3), dtype=np.float32)
        pred_kpts_viz[:, :2] = out['kp_np']              # (17, 2)

        img_rgb   = tensor_to_rgb(img_t)                  # HWC [0,1]
        h, w      = img_rgb.shape[:2]
        act_name  = C.ACT_CLASS_NAMES[act_cls]
        pred_name = C.ACT_CLASS_NAMES[out['act_pred']]
        cls_acc   = per_class_acc.get(act_cls, 0.0) * 100
        correct   = (out['act_pred'] == act_cls)

        # ── Left: Ground Truth ─────────────────────────────────────────────
        ax_gt = axes[row_i, 0]
        ax_gt.imshow(img_rgb, extent=[0, w, h, 0], aspect='auto')
        ax_gt.set_xlim(0, w); ax_gt.set_ylim(h, 0)
        if gt_boxes.shape[0] > 0:
            draw_det_boxes(ax_gt, gt_boxes, gt_labels)
        draw_skeleton(ax_gt, gt_kpts, conf_thresh=0.1, kp_color=KP_COLOR_GT)
        style_ax(
            ax_gt,
            f'GT  ▸  {act_name}\n'
            f'Class acc: {cls_acc:.1f}%   |   GT boxes: {gt_boxes.shape[0]}',
            color='#4ade80',
        )

        # ── Right: Prediction ──────────────────────────────────────────────
        ax_pr = axes[row_i, 1]
        ax_pr.imshow(img_rgb, extent=[0, w, h, 0], aspect='auto')
        ax_pr.set_xlim(0, w); ax_pr.set_ylim(h, 0)
        if len(pred_boxes) > 0:
            draw_det_boxes(ax_pr, pred_boxes, pred_labels, scores=pred_scores)
        draw_skeleton(ax_pr, pred_kpts_viz, conf_thresh=0.0,
                      kp_color=KP_COLOR_PRED, linewidth=1.5)
        verdict = '✓ CORRECT' if correct else '✗ WRONG'
        pred_color = '#4ade80' if correct else '#f87171'
        style_ax(
            ax_pr,
            f'PRED  ▸  {pred_name}   [{verdict}]\n'
            f'Pred boxes: {len(pred_boxes)}   (score>{SCORE_THRESH})',
            color=pred_color,
        )

    # Column headers
    axes[0, 0].set_title(
        f'GROUND TRUTH\n{axes[0, 0].get_title()}',
        fontsize=9, color='#4ade80', pad=6, fontweight='bold',
    )
    axes[0, 1].set_title(
        f'PREDICTION  (best.pth)\n{axes[0, 1].get_title()}',
        fontsize=9, color='#60a5fa', pad=6, fontweight='bold',
    )

    fig.suptitle(
        f'improved3 (GT-only) — Prediction vs Ground Truth\n'
        f'10 diverse activity classes  |  Overall val accuracy: {overall_acc*100:.1f}%  '
        f'|  mAP50: 0.591  |  PCK@0.05: 0.925',
        fontsize=12, color='white', y=1.001, fontweight='bold',
    )

    out_dir  = C.LOG_DIR
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / 'predictions_vs_gt.png'
    fig.savefig(out_path, dpi=120, bbox_inches='tight',
                facecolor=fig.get_facecolor())
    plt.close(fig)
    print(f'\nSaved: {out_path}')

    # ── Save accuracy table ───────────────────────────────────────────────────
    acc_path = out_dir / 'predictions_class_accuracy.txt'
    with open(acc_path, 'w') as f:
        f.write('Per-class activity accuracy — improved3 val set (best.pth)\n')
        f.write('=' * 65 + '\n')
        f.write(f'{"Class ID":>8}  {"Class Name":40s}  {"Correct":>7}  {"Total":>7}  {"Acc%":>7}\n')
        f.write('-' * 65 + '\n')
        for cls_id in sorted(class_total.keys()):
            name    = C.ACT_CLASS_NAMES[cls_id]
            correct = class_correct[cls_id]
            total   = class_total[cls_id]
            acc     = correct / total * 100 if total > 0 else 0.0
            f.write(f'{cls_id:>8}  {name:40s}  {correct:>7}  {total:>7}  {acc:>7.1f}%\n')
        total_correct = sum(class_correct.values())
        total_all     = sum(class_total.values())
        f.write('=' * 65 + '\n')
        f.write(f'{"":>8}  {"OVERALL":40s}  {total_correct:>7}  {total_all:>7}  '
                f'{total_correct / total_all * 100:>7.1f}%\n')

    print(f'Accuracy table: {acc_path}')


if __name__ == '__main__':
    main()
