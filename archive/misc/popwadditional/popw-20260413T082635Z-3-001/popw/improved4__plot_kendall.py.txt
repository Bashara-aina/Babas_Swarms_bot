import sys, os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))

"""
Plot Kendall Uncertainty Weight Evolution
==========================================
Reads metrics.jsonl and generates a two-panel figure:
  Left  : log(sigma^2) per task over epochs
  Right : Effective task weights over epochs

Usage:
    python plot_kendall.py
    python plot_kendall.py --log_path runs/ikea_multitask/logs/metrics.jsonl

Author: Bashara
Date: February 2026
"""

import argparse
import json
import logging
from pathlib import Path

import numpy as np

logger = logging.getLogger(__name__)


def main(args):
    logging.basicConfig(level=logging.INFO)

    log_path = Path(args.log_path)
    if not log_path.exists():
        logger.error(f'File not found: {log_path}')
        return

    epochs = []
    log_var_det, log_var_pose, log_var_act = [], [], []
    w_det, w_pose, w_act = [], [], []
    loss_det, loss_pose, loss_act, loss_total = [], [], [], []

    with open(log_path) as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            d = json.loads(line)

            # Kendall keys might be nested under 'train' or at top level
            train = d.get('train', d)

            if 'epoch' not in d:
                continue
            if 'log_var_det' not in train:
                continue

            epochs.append(d['epoch'])
            log_var_det.append(train['log_var_det'])
            log_var_pose.append(train['log_var_pose'])
            log_var_act.append(train['log_var_act'])
            w_det.append(train['w_det'])
            w_pose.append(train['w_pose'])
            w_act.append(train['w_act'])
            loss_det.append(train.get('det', 0))
            loss_pose.append(train.get('pose', 0))
            loss_act.append(train.get('activity', 0))
            loss_total.append(train.get('total', 0))

    if len(epochs) == 0:
        logger.error('No epochs with Kendall keys found in log file.')
        logger.info('Make sure train.py logs w_det/w_pose/w_act/log_var_* keys.')
        return

    logger.info(f'Found {len(epochs)} epochs with Kendall data.')

    try:
        import matplotlib
        matplotlib.use('Agg')
        import matplotlib.pyplot as plt
    except ImportError:
        logger.error('matplotlib not installed. Run: pip install matplotlib')
        return

    fig, axes = plt.subplots(2, 2, figsize=(14, 10))

    # Panel 1: log(sigma^2) evolution
    ax = axes[0, 0]
    ax.plot(epochs, log_var_det, label='det', linewidth=1.5)
    ax.plot(epochs, log_var_pose, label='pose', linewidth=1.5)
    ax.plot(epochs, log_var_act, label='activity', linewidth=1.5)
    ax.set_title('log(sigma^2) per task (Kendall)')
    ax.set_xlabel('Epoch')
    ax.set_ylabel('log(sigma^2)')
    ax.legend()
    ax.grid(True, alpha=0.3)

    # Panel 2: Effective task weights
    ax = axes[0, 1]
    ax.plot(epochs, w_det, label='w_det', linewidth=1.5)
    ax.plot(epochs, w_pose, label='w_pose', linewidth=1.5)
    ax.plot(epochs, w_act, label='w_act', linewidth=1.5)
    ax.set_title('Effective task weights')
    ax.set_xlabel('Epoch')
    ax.set_ylabel('Weight (normalized)')
    ax.legend()
    ax.grid(True, alpha=0.3)

    # Panel 3: Per-task losses
    ax = axes[1, 0]
    ax.plot(epochs, loss_det, label='det', linewidth=1.5)
    ax.plot(epochs, loss_pose, label='pose', linewidth=1.5)
    ax.plot(epochs, loss_act, label='activity', linewidth=1.5)
    ax.set_title('Per-task training loss')
    ax.set_xlabel('Epoch')
    ax.set_ylabel('Loss')
    ax.legend()
    ax.grid(True, alpha=0.3)

    # Panel 4: Total loss
    ax = axes[1, 1]
    ax.plot(epochs, loss_total, label='total', linewidth=1.5, color='black')
    ax.set_title('Total weighted loss')
    ax.set_xlabel('Epoch')
    ax.set_ylabel('Loss')
    ax.legend()
    ax.grid(True, alpha=0.3)

    plt.tight_layout()
    out_path = log_path.parent / 'kendall_weights.png'
    plt.savefig(out_path, dpi=150)
    plt.close()
    print(f'Saved: {out_path}')


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Plot Kendall Weight Evolution')
    parser.add_argument('--log_path', type=str,
                        default='runs/ikea_multitask/logs/metrics.jsonl',
                        help='Path to metrics.jsonl')
    main(parser.parse_args())
