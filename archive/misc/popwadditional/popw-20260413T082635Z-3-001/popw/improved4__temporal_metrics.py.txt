"""
Temporal Segmentation Metrics for IKEA Assembly
================================================
Computes segment-level metrics used by HVQ and View-Invariant papers:
  - F1@10, F1@25, F1@50  (segment-level F1 at IoU thresholds)
  - Edit Score           (normalized Levenshtein on segment sequences)

These require per-video frame-level predictions, so they run as a
standalone post-evaluation script rather than inside evaluate_all().

Usage:
    python temporal_metrics.py --checkpoint runs/ikea_multitask/checkpoints/best.pth
    python temporal_metrics.py --checkpoint best.pth --split test

Author: Bashara
Date: February 2026
"""

import logging
from collections import defaultdict
from pathlib import Path

import numpy as np
import torch
from torch.amp import autocast
from torch.utils.data import DataLoader

import config as C

from ikea_dataset import (
    IKEAMultiTaskDataset, collate_fn,
    _S_VIDEO_KEY, _S_FRAME_NUM,   # compact tuple index constants
)
from model import MultiTaskIKEA

logger = logging.getLogger(__name__)


# =============================================================================
# Core metric functions
# =============================================================================
def segment_bars(frame_labels):
    """Convert frame-level label array to list of (label, start, end) segments."""
    segs = []
    if len(frame_labels) == 0:
        return segs
    cur = frame_labels[0]
    start = 0
    for i in range(1, len(frame_labels)):
        if frame_labels[i] != cur:
            segs.append((int(cur), start, i - 1))
            cur = frame_labels[i]
            start = i
    segs.append((int(cur), start, len(frame_labels) - 1))
    return segs


def segment_iou(seg_a, seg_b):
    """IoU between two (label, start, end) segments. Returns 0 if different class."""
    if seg_a[0] != seg_b[0]:
        return 0.0
    inter_start = max(seg_a[1], seg_b[1])
    inter_end = min(seg_a[2], seg_b[2])
    inter = max(0, inter_end - inter_start + 1)
    len_a = seg_a[2] - seg_a[1] + 1
    len_b = seg_b[2] - seg_b[1] + 1
    union = len_a + len_b - inter
    return inter / union if union > 0 else 0.0


def f1_at_iou(gt_segs, pred_segs, iou_thresh):
    """Compute F1 score at a given IoU threshold between segment lists."""
    if len(pred_segs) == 0 and len(gt_segs) == 0:
        return 1.0
    if len(pred_segs) == 0 or len(gt_segs) == 0:
        return 0.0

    tp = 0
    fp = 0
    matched = set()
    for p in pred_segs:
        best_iou = 0.0
        best_j = -1
        for j, g in enumerate(gt_segs):
            if j in matched:
                continue
            iou = segment_iou(p, g)
            if iou > best_iou:
                best_iou = iou
                best_j = j
        if best_iou >= iou_thresh and best_j >= 0:
            tp += 1
            matched.add(best_j)
        else:
            fp += 1
    fn = len(gt_segs) - len(matched)
    prec = tp / (tp + fp + 1e-8)
    rec = tp / (tp + fn + 1e-8)
    return 2 * prec * rec / (prec + rec + 1e-8)


def edit_score(gt_labels, pred_labels):
    """Normalized Levenshtein distance on segment-label sequences."""
    gt_segs = segment_bars(gt_labels)
    pred_segs = segment_bars(pred_labels)
    gt_seq = [s[0] for s in gt_segs]
    pred_seq = [s[0] for s in pred_segs]

    m, n = len(gt_seq), len(pred_seq)
    if m == 0 and n == 0:
        return 1.0
    if m == 0 or n == 0:
        return 0.0

    dp = np.zeros((m + 1, n + 1), dtype=np.float64)
    for i in range(m + 1):
        dp[i, 0] = i
    for j in range(n + 1):
        dp[0, j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            cost = 0 if gt_seq[i - 1] == pred_seq[j - 1] else 1
            dp[i, j] = min(dp[i-1, j] + 1, dp[i, j-1] + 1, dp[i-1, j-1] + cost)
    return 1.0 - dp[m, n] / max(m, n)


def compute_all_temporal_metrics(gt_per_video, pred_per_video):
    """
    Compute temporal segmentation metrics across all videos.

    Args:
        gt_per_video   : list of np.ndarray -- frame-level GT labels per video
        pred_per_video : list of np.ndarray -- frame-level pred labels per video

    Returns:
        dict with F1@10, F1@25, F1@50, Edit
    """
    f1_10_list, f1_25_list, f1_50_list, edit_list = [], [], [], []
    for gt, pred in zip(gt_per_video, pred_per_video):
        gt_segs = segment_bars(gt)
        pred_segs = segment_bars(pred)
        f1_10_list.append(f1_at_iou(gt_segs, pred_segs, 0.10))
        f1_25_list.append(f1_at_iou(gt_segs, pred_segs, 0.25))
        f1_50_list.append(f1_at_iou(gt_segs, pred_segs, 0.50))
        edit_list.append(edit_score(gt, pred))
    return {
        'F1@10': float(np.mean(f1_10_list)) if f1_10_list else 0.0,
        'F1@25': float(np.mean(f1_25_list)) if f1_25_list else 0.0,
        'F1@50': float(np.mean(f1_50_list)) if f1_50_list else 0.0,
        'Edit': float(np.mean(edit_list)) if edit_list else 0.0,
    }


# =============================================================================
# Per-video inference
# =============================================================================
@torch.no_grad()
def predict_per_video(model, dataset, device, batch_size=C.BATCH_SIZE):
    """
    Run inference and group frame-level predictions by video.

    NOTE: dataset.samples is a list of compact tuples — use the
    _S_VIDEO_KEY / _S_FRAME_NUM index constants, NOT dict keys.

    Returns:
        gt_per_video   : list of np.ndarray
        pred_per_video : list of np.ndarray
    """
    model.eval()
    loader = DataLoader(
        dataset, batch_size=batch_size, shuffle=False,
        num_workers=C.NUM_WORKERS, collate_fn=collate_fn,
        pin_memory=C.PIN_MEMORY,
    )

    # Collect all predictions ordered by dataset index
    all_preds = []
    all_gts = []
    all_video_keys = []

    sample_idx = 0
    for images, targets in loader:
        images = _prepare_images(images, device)
        with autocast(device_type='cuda', enabled=C.MIXED_PRECISION):
            outputs = model(images)
        preds = outputs['act_logits'].argmax(dim=1).cpu().numpy()
        gts = targets['activity'].numpy()

        for i in range(len(preds)):
            if sample_idx < len(dataset.samples):
                all_preds.append(int(preds[i]))
                all_gts.append(int(gts[i]))
                # FIX: samples is a compact tuple — use integer index constant
                all_video_keys.append(dataset.samples[sample_idx][_S_VIDEO_KEY])
            sample_idx += 1

    # Group by video, preserving frame order
    video_gt = defaultdict(list)
    video_pred = defaultdict(list)

    for i in range(len(all_preds)):
        if i < len(dataset.samples):
            vk = all_video_keys[i]
            # FIX: use _S_FRAME_NUM integer index, not dict key 'frame_num'
            fn = dataset.samples[i][_S_FRAME_NUM]
            video_gt[vk].append((fn, all_gts[i]))
            video_pred[vk].append((fn, all_preds[i]))

    gt_per_video = []
    pred_per_video = []
    for vk in sorted(video_gt.keys()):
        gt_sorted = sorted(video_gt[vk], key=lambda x: x[0])
        pred_sorted = sorted(video_pred[vk], key=lambda x: x[0])
        gt_per_video.append(np.array([g for _, g in gt_sorted]))
        pred_per_video.append(np.array([p for _, p in pred_sorted]))

    return gt_per_video, pred_per_video


# =============================================================================
# CLI
# =============================================================================
def main(args):
    logging.basicConfig(
        level=logging.INFO,
        format='%(asctime)s | %(levelname)s | %(message)s',
    )

    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    logger.info(f'Device: {device}')

    # Load dataset
    logger.info(f'Loading {args.split} split ...')
    ds = IKEAMultiTaskDataset(split=args.split, augment=False)
    # FIX: use _S_VIDEO_KEY index constant — samples are tuples, not dicts
    n_videos = len(set(s[_S_VIDEO_KEY] for s in ds.samples))
    logger.info(f'  {len(ds):,} frames, {n_videos} videos')

    # Load model
    logger.info(f'Loading checkpoint: {args.checkpoint}')
    model = MultiTaskIKEA(pretrained=False, use_film=C.USE_FILM).to(device)
    ckpt = torch.load(args.checkpoint, map_location=device)
    model.load_state_dict(ckpt['model'], strict=False)
    logger.info(f'  Loaded from epoch {ckpt.get("epoch", "?")}')

    # Run per-video inference
    logger.info('Running per-video inference ...')
    gt_per_video, pred_per_video = predict_per_video(model, ds, device)
    logger.info(f'  {len(gt_per_video)} videos processed')

    # Compute temporal metrics
    metrics = compute_all_temporal_metrics(gt_per_video, pred_per_video)

    print('\n' + '=' * 50)
    print(f'Temporal Segmentation Metrics ({args.split})')
    print('=' * 50)
    print(f'  F1@10      : {metrics["F1@10"]:.4f}')
    print(f'  F1@25      : {metrics["F1@25"]:.4f}')
    print(f'  F1@50      : {metrics["F1@50"]:.4f}')
    print(f'  Edit Score : {metrics["Edit"]:.4f}')
    print('=' * 50)


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='Temporal Segmentation Metrics')
    parser.add_argument('--checkpoint', type=str, required=True)
    parser.add_argument('--split', type=str, default='test')
    main(parser.parse_args())
