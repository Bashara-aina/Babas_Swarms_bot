import argparse
import json
import statistics
import threading
import time
from dataclasses import dataclass
from pathlib import Path
from typing import Dict, List, Optional, Tuple

import numpy as np
import psutil
import torch
from torch.optim import AdamW
from torch.utils.data import DataLoader

import config as C
from ikea_dataset import IKEAMultiTaskDataset, collate_fn
from losses import MultiTaskLoss
from model import MultiTaskIKEA


@dataclass
class SweepConfig:
    batch_size: int
    num_workers: int
    prefetch_factor: int

    @property
    def key(self) -> str:
        return f"bs{self.batch_size}_nw{self.num_workers}_pf{self.prefetch_factor}"


class ResourceSampler:
    def __init__(self, interval_sec: float = 0.5) -> None:
        self.interval_sec = interval_sec
        self._stop_event = threading.Event()
        self._thread: Optional[threading.Thread] = None

        self.cpu_util_samples: List[float] = []
        self.sys_ram_used_gb_samples: List[float] = []
        self.proc_rss_gb_samples: List[float] = []
        self.gpu_util_samples: List[float] = []
        self.gpu_mem_gb_samples: List[float] = []

        self._nvml_available = False
        self._nvml_handle = None
        try:
            import pynvml  # type: ignore

            pynvml.nvmlInit()
            self._nvml_handle = pynvml.nvmlDeviceGetHandleByIndex(0)
            self._nvml = pynvml
            self._nvml_available = True
        except Exception:
            self._nvml_available = False
            self._nvml = None

    def start(self) -> None:
        self._thread = threading.Thread(target=self._run, daemon=True)
        self._thread.start()

    def stop(self) -> None:
        self._stop_event.set()
        if self._thread is not None:
            self._thread.join(timeout=2.0)
        if self._nvml_available and self._nvml is not None:
            try:
                self._nvml.nvmlShutdown()
            except Exception:
                pass

    def _run(self) -> None:
        proc = psutil.Process()
        while not self._stop_event.is_set():
            self.cpu_util_samples.append(psutil.cpu_percent(interval=None))
            self.sys_ram_used_gb_samples.append(psutil.virtual_memory().used / 1e9)
            self.proc_rss_gb_samples.append(proc.memory_info().rss / 1e9)

            if self._nvml_available and self._nvml is not None and self._nvml_handle is not None:
                try:
                    util = self._nvml.nvmlDeviceGetUtilizationRates(self._nvml_handle)
                    mem = self._nvml.nvmlDeviceGetMemoryInfo(self._nvml_handle)
                    self.gpu_util_samples.append(float(util.gpu))
                    self.gpu_mem_gb_samples.append(mem.used / 1e9)
                except Exception:
                    pass

            time.sleep(self.interval_sec)


def parse_sweep(raw: str) -> List[SweepConfig]:
    configs: List[SweepConfig] = []
    for chunk in raw.split(','):
        chunk = chunk.strip()
        if not chunk:
            continue
        try:
            bs_str, nw_str, pf_str = chunk.split('-')
            configs.append(
                SweepConfig(
                    batch_size=int(bs_str),
                    num_workers=int(nw_str),
                    prefetch_factor=int(pf_str),
                )
            )
        except Exception as exc:
            raise ValueError(
                f"Invalid sweep item '{chunk}'. Expected format 'batch-workers-prefetch'."
            ) from exc
    if not configs:
        raise ValueError("No valid sweep configs provided")
    return configs


def build_loader(ds: IKEAMultiTaskDataset, cfg: SweepConfig) -> DataLoader:
    return DataLoader(
        ds,
        batch_size=cfg.batch_size,
        sampler=ds.get_sampler(),
        shuffle=False,
        num_workers=cfg.num_workers,
        collate_fn=collate_fn,
        pin_memory=C.PIN_MEMORY,
        drop_last=True,
        persistent_workers=(cfg.num_workers > 0),
        prefetch_factor=cfg.prefetch_factor if cfg.num_workers > 0 else None,
    )


def prepare_images(images: torch.Tensor, device: torch.device) -> torch.Tensor:
    images = images.to(device, non_blocking=True)
    if images.dtype == torch.uint8:
        images = images.float().div_(255.0)
        mean = torch.tensor(C.IMAGENET_MEAN, device=device, dtype=images.dtype).view(1, 3, 1, 1)
        std = torch.tensor(C.IMAGENET_STD, device=device, dtype=images.dtype).view(1, 3, 1, 1)
        images = (images - mean) / std
    return images


def move_targets_to_device(targets: Dict, device: torch.device) -> Dict:
    for index in range(len(targets['detection'])):
        targets['detection'][index]['boxes'] = targets['detection'][index]['boxes'].to(device)
        targets['detection'][index]['labels'] = targets['detection'][index]['labels'].to(device)
    targets['keypoints'] = targets['keypoints'].to(device)
    targets['visibility'] = targets['visibility'].to(device)
    targets['kpt_confidence'] = targets['kpt_confidence'].to(device)
    targets['activity'] = targets['activity'].to(device)
    return targets


def aggregate(samples: List[float]) -> float:
    return float(statistics.mean(samples)) if samples else 0.0


def run_single_config(
    cfg: SweepConfig,
    train_ds: IKEAMultiTaskDataset,
    model: MultiTaskIKEA,
    criterion: MultiTaskLoss,
    optimizer: torch.optim.Optimizer,
    device: torch.device,
    warmup_steps: int,
    measured_steps: int,
) -> Dict:
    loader = build_loader(train_ds, cfg)
    loader_iter = iter(loader)

    torch.cuda.reset_peak_memory_stats(device)

    timer_samples: List[float] = []
    sampler = ResourceSampler(interval_sec=0.4)
    sampler.start()

    success = True
    error_message = ''

    total_steps = warmup_steps + measured_steps

    try:
        model.train()
        optimizer.zero_grad(set_to_none=True)

        for step_idx in range(total_steps):
            try:
                images, targets = next(loader_iter)
            except StopIteration:
                loader_iter = iter(loader)
                images, targets = next(loader_iter)

            start_t = time.perf_counter()

            images = prepare_images(images, device)
            targets = move_targets_to_device(targets, device)

            with torch.amp.autocast(device_type='cuda', enabled=C.MIXED_PRECISION):
                outputs = model(images)

            for key in ('cls_preds', 'reg_preds', 'keypoints', 'act_logits'):
                if key in outputs and isinstance(outputs[key], torch.Tensor):
                    outputs[key] = outputs[key].float()

            loss, _ = criterion(outputs, targets)
            if not torch.isfinite(loss):
                continue

            loss.backward()
            optimizer.step()
            optimizer.zero_grad(set_to_none=True)

            torch.cuda.synchronize(device)
            elapsed = time.perf_counter() - start_t

            if step_idx >= warmup_steps:
                timer_samples.append(elapsed)

    except RuntimeError as exc:
        success = False
        error_message = str(exc)
    finally:
        sampler.stop()
        del loader

    throughput = 0.0
    if timer_samples:
        mean_step_time = float(np.mean(timer_samples))
        throughput = cfg.batch_size / mean_step_time if mean_step_time > 0 else 0.0

    return {
        'config': cfg.__dict__,
        'success': success,
        'error': error_message,
        'measured_steps': len(timer_samples),
        'throughput_img_s': throughput,
        'step_time_ms': (float(np.mean(timer_samples)) * 1000.0) if timer_samples else 0.0,
        'gpu_peak_mem_alloc_gb': torch.cuda.max_memory_allocated(device) / 1e9,
        'gpu_peak_mem_reserved_gb': torch.cuda.max_memory_reserved(device) / 1e9,
        'cpu_util_avg_pct': aggregate(sampler.cpu_util_samples),
        'sys_ram_used_avg_gb': aggregate(sampler.sys_ram_used_gb_samples),
        'proc_rss_avg_gb': aggregate(sampler.proc_rss_gb_samples),
        'gpu_util_avg_pct': aggregate(sampler.gpu_util_samples),
        'gpu_mem_avg_gb': aggregate(sampler.gpu_mem_gb_samples),
    }


def pick_best(results: List[Dict]) -> Optional[Dict]:
    ok = [r for r in results if r['success'] and r['measured_steps'] > 0]
    if not ok:
        return None
    ok.sort(key=lambda item: item['throughput_img_s'], reverse=True)
    return ok[0]


def main() -> None:
    parser = argparse.ArgumentParser(description='Micro benchmark for training sweet-spot search')
    parser.add_argument(
        '--sweep',
        type=str,
        default='10-0-1,10-1-2,10-2-2,12-1-2,12-2-2,14-1-2',
        help='Comma-separated configs: batch-workers-prefetch',
    )
    parser.add_argument('--warmup-steps', type=int, default=10)
    parser.add_argument('--measured-steps', type=int, default=50)
    parser.add_argument(
        '--output',
        type=str,
        default=str(Path(C.LOG_DIR) / 'micro_benchmark_results.json'),
    )
    args = parser.parse_args()

    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    if device.type != 'cuda':
        raise RuntimeError('CUDA is required for this benchmark')

    torch.backends.cudnn.benchmark = bool(getattr(C, 'CUDNN_BENCHMARK', True))
    torch.backends.cudnn.deterministic = bool(getattr(C, 'CUDNN_DETERMINISTIC', False))
    torch.backends.cuda.matmul.allow_tf32 = bool(getattr(C, 'ALLOW_TF32', True))
    torch.backends.cudnn.allow_tf32 = bool(getattr(C, 'ALLOW_TF32', True))
    torch.set_float32_matmul_precision(str(getattr(C, 'MATMUL_PRECISION', 'high')))

    sweep_configs = parse_sweep(args.sweep)

    print(f"Using config: {Path(C.__file__).resolve()}")
    print(f"Sweep configs: {[cfg.key for cfg in sweep_configs]}")

    train_ds = IKEAMultiTaskDataset(split='train', img_size=C.IMG_SIZE, augment=True, seed=C.SEED)

    model = MultiTaskIKEA(pretrained=True, use_film=C.USE_FILM).to(device)
    criterion = MultiTaskLoss(
        num_classes=C.NUM_ACT_CLASSES,
        train_det=C.TRAIN_DET,
        train_pose=C.TRAIN_POSE,
        train_act=C.TRAIN_ACT,
        use_kendall=C.USE_KENDALL,
    ).to(device)
    criterion.set_class_counts(train_ds.class_counts)

    optimizer = AdamW(model.parameters(), lr=C.BASE_LR, weight_decay=C.WEIGHT_DECAY)

    results: List[Dict] = []
    for cfg in sweep_configs:
        print(f"\n=== Running {cfg.key} ===")
        run_result = run_single_config(
            cfg=cfg,
            train_ds=train_ds,
            model=model,
            criterion=criterion,
            optimizer=optimizer,
            device=device,
            warmup_steps=args.warmup_steps,
            measured_steps=args.measured_steps,
        )
        results.append(run_result)
        print(json.dumps(run_result, indent=2))

        if torch.cuda.is_available():
            torch.cuda.empty_cache()

    best = pick_best(results)

    payload = {
        'config_path': str(Path(C.__file__).resolve()),
        'timestamp': time.strftime('%Y-%m-%d %H:%M:%S'),
        'results': results,
        'best': best,
    }

    output_path = Path(args.output)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(json.dumps(payload, indent=2), encoding='utf-8')

    print('\n=== BEST CONFIG ===')
    if best is None:
        print('No successful configuration found.')
    else:
        print(json.dumps(best, indent=2))
    print(f"Saved benchmark report to: {output_path}")


if __name__ == '__main__':
    main()
